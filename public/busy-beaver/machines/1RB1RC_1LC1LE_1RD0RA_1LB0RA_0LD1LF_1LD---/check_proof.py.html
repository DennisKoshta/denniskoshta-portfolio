# ---------------------------------------------------------------- A. chunk lemmas
def chunk(before, bstate, bhead, after, astate, ahead, nsteps):
    """Run nsteps from (before, bstate on index bhead). The head may not leave the window except
    with the very last move. Check the window equals `after` and the head is on index ahead."""
    s = Sim(before, head=bhead, state=bstate, pad=8)
    lo, hi = s.off, s.off + len(before)
    for k in range(nsteps):
        if not (lo <= s.pos < hi):
            return False
        s.run(1)
        if s.halted:
            return False
    return s.window(0, len(before)) == after and s.state == astate and s.pos - s.off == ahead


def sec_A():
    ok = True
    # cells written left to right; heads are window indices; -1 means "the cell left of the window"
    lem = {
        'R  0011 D>0      -> D>0 1111     (4)': ('00110', 'D', 4, '01111', 'D', 0, 4),
        'S  x101 D>0      -> D>x 1111     (8)': (None, 'D', 4, None, 'D', 0, 8),
        'P  001 D>0       -> D>0 011      (3)': ('0010', 'D', 3, '0011', 'D', 0, 3),
        'T  10 D>0        -> 010 A>       (5)': ('1001', 'D', 2, '0101', 'A', 3, 5),
        'A  A>11          -> 10 A>        (2)': ('11', 'A', 0, '10', 'A', 2, 2),
        'H  10 A>0 1      -> 010 A>1      (7)': ('1001', 'A', 2, '0101', 'A', 3, 7),
        'E  10 A>000      -> 1 0011 D>0   (5)': ('10000', 'A', 2, '10011', 'D', 5, 5),
    }
    for name, (b, bs, bh, a, as_, ah, n) in lem.items():
        if name.startswith('S'):
            good = all(chunk(x + '1010', 'D', 4, x + '1111', 'D', 0, 8) for x in '01')
        elif name.startswith('E'):
            s = Sim('10000000', head=2, state='A', pad=8); s.run(5)
            good = s.window(0, 8) == '10011000' and s.state == 'D' and s.pos - s.off == 5
        else:
            good = chunk(b, bs, bh, a, as_, ah, n)
        print('   ', name, 'ok' if good else 'FAIL')
        ok &= good
    # inductive corollaries, checked for small sizes: even field sweep and odd field sweep
    for f in range(0, 30):
        cells = '00' + '1' + '01' * f + '0' + '1' * 6   # sep | field | D-cell | 1^6
        s = Sim(cells, head=2 + 2 * f + 1, state='D', pad=8)
        s.run(4 * f + (4 if f % 2 else 0))
        if f % 2 == 0:
            exp = '00' + '1' + '0' + '1' * (2 * f) + '1' * 6
            good = s.window(0, len(cells)) == exp and s.state == 'D' and s.pos - s.off == 3
        else:
            exp = '00' + '1' * (2 * f + 2) + '1' * 6
            good = s.window(0, len(cells)) == exp and s.state == 'D' and s.pos - s.off == 1
        ok &= good
    report('A', ok, '7 chunk lemmas exact; even/odd field sweeps (4f, 4f+4 steps) for f = 0..29')


# ---------------------------------------------------------------- B. macro rules
def rnd_even(lo=2, hi=40): return 2 * R.randint(lo // 2, hi // 2)
def rnd_odd(lo=1, hi=41): return 2 * R.randint((lo - 1) // 2, (hi - 1) // 2) + 1
def rnd(lo=1, hi=40): return R.randint(lo, hi)


def seq(fs, ss=None):
    ss = ss if ss is not None else [PIPE] * (len(fs) - 1)
    return mk(fs, ss)


def gen(rule):
    """A random instance on which the rule's case must fire (left context random)."""
    left = [rnd(3, 40) for _ in range(R.randint(1, 3))]
    lsep = [R.choice([PIPE, PIPE, PLUS]) for _ in left[1:]]
    evens = [rnd_even(2, 40) for _ in range(R.randint(0, 3))]
    L = rnd_even(2, 60)
    if rule == 'odd':
        d = rnd(1, 40); j = rnd_odd(1, 41)
        return mk(left + [d, j] + evens + [L], lsep + [R.choice([PIPE, PLUS]), PIPE] + [PIPE] * (len(evens) + 1))
    if rule == 'odd-L':
        d = rnd(1, 40); Lo = rnd_odd(1, 61)
        return mk(left + [d, Lo], lsep + [R.choice([PIPE, PLUS]), PIPE])
    if rule == 'plus':
        x = rnd(2, 40)
        return mk(left + [x] + evens + [L], lsep + [R.choice([PIPE, PLUS])] + [PLUS] + [PIPE] * len(evens))
    if rule == 'hash-odd':
        a = rnd(1, 40)
        return mk(left + [a] + evens + [L], lsep + [R.choice([PIPE, PLUS])] + [HASH] + [PIPE] * len(evens))
    if rule == 'death':
        b = rnd_odd(1, 41)
        return mk(left + [0, b] + evens + [L], lsep + [PIPE, PIPE] + [PIPE] * (len(evens) + 1))
    if rule == 'death-merge':
        b = rnd_odd(1, 41); ev = evens or [rnd_even()]
        return mk(left + [0, b] + ev + [L], lsep + [PLUS, PIPE] + [PIPE] * (len(ev) + 1))
    if rule == 'death-merge-far':
        b = rnd_odd(1, 41); ev = evens or [rnd_even()]
        return mk([1, 0, b] + ev + [L], [PLUS, PIPE] + [PIPE] * (len(ev) + 1))
    if rule == 'odd-plus':
        d = rnd(1, 40); j = rnd_odd(1, 41)
        return mk(left + [d, j] + evens + [L], lsep + [R.choice([PIPE, PLUS]), PLUS] + [PIPE] * (len(evens) + 1))
    if rule == 'bounce':
        a = rnd(0, 40); b = rnd_odd(1, 41)
        return mk(left + [a, b] + evens + [L], lsep + [R.choice([PIPE, PLUS]), HASH] + [PIPE] * (len(evens) + 1))
    if rule == 'bounce-L':
        a = rnd(0, 40); Lo = rnd_odd(1, 61)
        return mk(left + [a, Lo], lsep + [R.choice([PIPE, PLUS]), HASH])
    if rule == 'overflow-even':
        fs = [rnd_even(0, 40)] + evens + [L]
        return seq(fs)
    if rule == 'overflow-odd':
        fs = [rnd_odd(1, 41)] + evens + [L]
        return seq(fs)
    if rule == 'plus-farend-1':
        return mk([1] + evens + [L], [PLUS] + [PIPE] * len(evens))
    if rule == 'plus-merge-1':
        y = rnd(1, 40); e = rnd_even(0, 40)
        return mk(left + [y, 1, e] + evens + [L], lsep + [R.choice([PIPE, PLUS]), PLUS, PLUS] + [PIPE] * (len(evens) + 1))
    if rule == 'death-farend':
        b = rnd_odd(1, 41)
        return mk([0, b] + evens + [L], [PIPE, PIPE] + [PIPE] * len(evens))
    if rule == 'hash-farend-0':
        e = rnd_even(0, 40)
        return mk([0, e] + evens + [L], [HASH, PIPE] + [PIPE] * len(evens))
    if rule == 'far0plus-even':
        return mk([0] + evens + [L], [PLUS] + [PIPE] * len(evens))
    if rule == 'far0plus-odd':
        j = rnd_odd(1, 41)
        if R.random() < 0.25:
            return mk([0, j], [PLUS])
        return mk([0, j] + evens + [L], [PLUS, PIPE] + [PIPE] * len(evens))
    raise ValueError(rule)

# ---------------------------------------------------------------- D. counter lemma
def region_state(fs, L):
    """B bits (MSB first), N, P for a region g_1..g_n | L."""
    B = []; acc = 0
    for g in fs:
        acc ^= g & 1; B.append(acc)
    P = acc ^ (L & 1)
    N = 0
    for b in B:
        N = 2 * N + b
    return B, N, P


def sec_D():
    ok = True
    # bit level, exhaustive for n <= 10: the Gray-scan step equals N -> N +/- 1 with the stated carry depth
    for n in range(1, 11):
        for bits in range(1 << (n + 1)):
            b = [(bits >> (n - i)) & 1 for i in range(n + 1)]      # b_1..b_n, b_L
            fs = b[:n]; L = b[n]
            Bv, N, P = region_state(fs, L)
            if (N == 0 and P == 0) or (N == (1 << n) - 1 and P == 1):
                continue
            j = max(i for i in range(n + 1) if b[i] == 1)        # 0-based; n means L
            if j == 0:
                ok = False; continue
            nb = list(b); nb[j - 1] ^= 1; nb[n] ^= 1
            B2, N2, P2 = region_state(nb[:n], nb[n])
            dep = n - 1 - (j - 1)                                 # depth of the decremented field
            if P == 1:
                good = N2 == N + 1 and P2 == 1 and dep == ((N + 1) & -(N + 1)).bit_length() - 1
            else:
                good = N2 == N - 1 and P2 == 0 and dep == (N & -N).bit_length() - 1
            ok &= good
    # on the abstract map: random regions with large fields behind a '+', '#' or the far end
    cnt = 0
    for _ in range(300 if QUICK else 2000):
        n = R.randint(1, 7)
        fs = [R.randint(2 ** n + 2, 2 ** n + 60) for _ in range(n)]
        L = R.randint(1, 50)
        left = R.choice(['far', 'plus', 'hash'])
        pre = [] if left == 'far' else [R.randint(5, 40), PLUS if left == 'plus' else HASH]
        T = pre + seq(fs + [L])
        Bv, N, P = region_state(fs, L)
        if (N == 0 and P == 0) or (N == (1 << n) - 1 and P == 1):
            continue
        U, st, rule = step(T)
        fs2 = U[len(pre)::2][:n]; L2 = U[-1]
        B2, N2, P2 = region_state(fs2, L2)
        dec = [a - b for a, b in zip(fs, fs2)]
        v = ((N + 1) & -(N + 1)).bit_length() - 1 if P == 1 else (N & -N).bit_length() - 1
        exp = [1 if (n - 1 - i) == v else 0 for i in range(n)]
        good = rule == 'odd' and N2 == N + (1 if P else -1) and P2 == P and dec == exp and L2 == L + 3
        ok &= good; cnt += 1
    report('D', ok, f'counter lemma exhaustive on bits (n <= 10) and on {cnt} random regions of the abstract map')

def opening(start, direct_limit):
    s = Sim('', head=0, state=start)
    log = []

    def stop(sm):
        U = parse_at(sm)
        if U is not None:
            log.append((sm.steps, U))
            if sm.steps >= BASES[start][0]:
                return True
        return False
    s.run(BASES[start][0] + 10, stop)
    first = log[0]
    base = log[-1]
    ok = base[0] == BASES[start][0] and base[1] == [BASES[start][1]] + BASE_REST
    four = [x for x in log if 4 in x[1][1::2]]
    # from the base: predicted D:R configurations must be the actual ones
    T = base[1]; t = base[0]; npred = 0; rules = Counter()
    target = [t]

    def stop2(sm):
        return sm.steps == target[0] and parse_at(sm) == target[1]
    while True:
        P, st, rule = step(T)
        if P is None or t + st > direct_limit:
            break
        target[:] = [t + st, P]
        hit = s.run(t + st - s.steps, stop2) if s.steps < t + st else False
        if not hit:
            ok = False
            print('    prediction failed at', t, fmt(T), rule)
            break
        T = P; t += st; npred += 1; rules[rule] += 1
    return ok, first, four, base, npred, t, rules, T


def relabel(tm, perm, mirror):
    """perm maps old state letter -> new letter; returns the table string in the new names."""
    t = parse_tm(tm)
    inv = {v: k for k, v in perm.items()}
    out = []
    for q in ST:
        old = inv[q]
        g = ''
        for sym in (0, 1):
            tr = t[(old, sym)]
            if tr is None:
                g += '---'
            else:
                w, d, nq = tr
                d = -d if mirror else d
                g += f'{w}{"R" if d == 1 else "L"}{perm[nq]}'
        out.append(g)
    return '_'.join(out)

# ---------------------------------------------------------------- G. invariant J
def check_J(T):
    """Condition J of proof.md section 6.1 (revision 2: J3 allows a first field 0 before '+').
    Returns None if J holds, else the failing clause."""
    fs, ss = T[0::2], T[1::2]
    k = len(fs) - 1
    if fs[-1] < 1 or any(f < 0 for f in fs):
        return 'J0 fields >= 0, L >= 1'
    if any(s not in (PIPE, PLUS, HASH) for s in ss):
        return 'J1 separators | + #'
    hs = [i for i, s in enumerate(ss) if s == HASH]
    if len(hs) > 1:
        return 'J2 at most one #'
    if hs:
        h = hs[0]
        if any(s != PIPE for s in ss[h + 1:]):
            return 'J2 only | right of #'
        if fs[h + 1] < 1:
            return 'J2 field right of # >= 1'
        if fs[h] < 1 and h != 0:
            return 'J2 field left of # >= 1 unless first'
    for i, s in enumerate(ss):
        if s == PLUS:
            if fs[i] == 0 and i != 0:
                return 'J3 field left of + >= 1 unless first'
            if fs[i] == 1 and i != 0 and ss[i - 1] != PLUS:
                return 'J3 a 1 left of + is first or follows +'
    if k >= 1 and ss[k - 1] == PLUS:
        return 'J4 separator left of L is | or #'
    if k >= 2 and fs[k - 1] == 0 and ss[k - 2] == PLUS:
        return 'J5 no + 0 | L'
    return None


def legal_event_exhaustive():
    """Legal-event lemma: every token list satisfying J has a rule (step(T) is not None).
    Exhaustive over <= 5 fields with values 0..4 and separators | + #."""
    import itertools
    n = 0
    for nf in range(1, 6 if not QUICK else 5):
        for fs in itertools.product(range(5), repeat=nf):
            for ss in itertools.product((PIPE, PLUS, HASH), repeat=nf - 1):
                T = mk(list(fs), list(ss))
                if check_J(T) is None:
                    n += 1
                    if step(T)[0] is None:
                        print('    legal-event lemma fails at', fmt(T), step(T)[2])
                        return False, n
    return True, n


def exit_class(T, rule, why):
    """Label a one-step exit from J by the classes (a)-(e) of proof.md 6.3."""
    if rule == 'hash-odd':
        return 'a'
    if rule == 'bounce':
        return 'b'
    if rule == 'odd' and why.startswith('J2'):
        return 'c'
    if rule == 'death':
        return 'd'
    if rule == 'odd' and why.startswith('J5'):
        return 'e'
    return '?'


def one_step_catalogue():
    import itertools
    cat = {}; nj = 0
    for nf in range(1, 6 if not QUICK else 5):
        for fs in itertools.product(range(5), repeat=nf):
            for ss in itertools.product((PIPE, PLUS, HASH), repeat=nf - 1):
                T = mk(list(fs), list(ss))
                if check_J(T) is not None:
                    continue
                nj += 1
                P, st, rule = step(T)
                why = check_J(P)
                if why is None:
                    continue
                key = (exit_class(T, rule, why), rule, why)
                c, ex = cat.get(key, (0, None))
                if ex is None or sum(T[0::2]) < sum(e for e in map(int, ex.split()[0::2])):
                    ex = fmt(T)
                cat[key] = (c + 1, ex)
    return cat, nj

# ---------------------------------------------------------------- H. death / dance / climb / reflection
def climb_predict(T, maxstages=400):
    """The climb automaton of proof.md 4.5 (C5), from C2 and the stage-end events of 4.3 only.
    Active region g_1 | ... | g_m | L with '+' or the far end left of g_1, g_2.. even, L odd. Mode: g_1 odd -> down-run
    from 2^m - 1 (D); g_1 even -> up-run from 0 (U). Returns ([(half-cycles so far, predicted list, event)], end)."""
    fs, ss = list(T[0::2]), list(T[1::2])
    k = len(fs) - 1
    lo = k - 1
    while lo > 0 and ss[lo - 1] == PIPE:
        lo -= 1
    h = 0; out = []
    for _ in range(maxstages):
        m = k - lo
        if not (all(f % 2 == 0 for f in fs[lo + 1:k]) and fs[k] % 2 == 1):
            return out, 'not a climb region'
        mode = 'D' if fs[lo] % 2 else 'U'
        if mode == 'U' and fs[lo] == 0:
            nst = (1 << (m - 1)) - 1             # up-run to the first carry into g_1 = 0
            for u in range(m - 1):
                fs[k - 1 - u] -= 1 << (m - 2 - u)
            fs[k] += 3 * nst; h += nst
            if any(f < 0 for f in fs[lo:k]):
                return out, 'death-in-stage'
            if lo == 0:
                del fs[0]; del ss[0]; k -= 1; fs[k] += 3; h += 1
                out.append((h, mk(fs, ss), 'death-farend')); continue
            if m >= 2 and ss[lo - 1] == PLUS:
                y = fs[lo - 1]; j = fs[lo + 1]
                fs[lo - 1:lo + 2] = [y + j + 1]; del ss[lo - 1:lo + 1]; k -= 2; lo -= 1; fs[k] += 3; h += 1
                out.append((h, mk(fs, ss), 'death-merge'))
                if lo > 0 and ss[lo - 1] != PLUS:
                    return out, 'complete'
                continue
            return out, 'outside'
        nst = (1 << m) - 1
        for u in range(m):
            fs[k - 1 - u] -= 1 << (m - 1 - u)
        fs[k] += 3 * nst; h += nst
        if any(f < 0 for f in fs[lo:k]):
            return out, 'death-in-stage'
        if lo == 0:
            return out, 'far end: overflow-%s next' % ('even' if mode == 'D' else 'odd')
        if ss[lo - 1] != PLUS:
            return out, 'complete'
        z = fs[lo - 1]; zl = ss[lo - 2] if lo >= 2 else None
        if mode == 'D':
            if z >= 2:
                fs[lo - 1] -= 2; ss[lo - 1] = PIPE; lab = 'plus'
            elif z == 1 and zl is None:
                del fs[0]; del ss[0]; k -= 1; lo -= 1; fs[k] += 3; h += 1
                out.append((h, mk(fs, ss), 'plus-farend-1')); continue
            elif z == 1 and zl == PLUS:
                y = fs[lo - 2]; g1 = fs[lo]
                fs[lo - 2:lo + 1] = [y + g1 + 1]; del ss[lo - 2:lo]; k -= 2; lo -= 1; lab = 'plus-merge-1'
            elif z == 0 and zl is None:
                return out, 'far0plus-even next'
            else:
                return out, 'outside'
        else:
            if z >= 1:
                fs[lo - 1] -= 1; fs[lo] -= 1; ss[lo - 1] = PIPE; lab = 'odd-plus'
            elif zl is None:
                return out, 'far0plus-odd next'
            else:
                return out, 'outside'
        lo -= 1; fs[k] += 3; h += 1
        out.append((h, mk(fs, ss), lab))
        if lo > 0 and ss[lo - 1] != PLUS:
            return out, 'complete'
    return out, 'maxstages'


def climb_check(T, seq=None):
    """Compare climb_predict(T) with the abstract map, half-cycle by half-cycle (seq: the map's own lists after T,
    if known). Returns (status, number of stage ends matched, predicted events, end)."""
    pred, end = climb_predict(T)
    cur = T; h = 0; nst = 0; last = None
    for (hp, Tp, lab) in pred:
        while h < hp:
            if seq is not None:
                if h >= len(seq):
                    return 'short', nst, pred, end
                last, cur = seq[h]
            else:
                P_, st_, last = step(cur)
                if P_ is None:
                    return 'outside ' + last, nst, pred, end
                cur = P_
            h += 1
            if h < hp and last != 'odd':
                return ('nested ' if last.startswith('death') else 'MISMATCH unpredicted ') + last, nst, pred, end
        if cur != Tp or last != lab:
            return 'MISMATCH at %d: %s %s vs %s %s' % (h, last, fmt(cur)[:60], lab, fmt(Tp)[:60]), nst, pred, end
        nst += 1
    return 'ok', nst, pred, end


def region_of(fs, ss, idx):
    """First index of the region that contains field idx."""
    lo = idx
    while lo > 0 and ss[lo - 1] == PIPE:
        lo -= 1
    return lo


def sec_H(far):
    """C4 (dance), C5 (climb automaton) and C6 (reflection, both branches) on the four orbits to `far`, plus random
    climbs with singletons equal to 1."""
    ok = True
    REPAIR = {'bounce', 'overflow-even', 'overflow-odd', 'far0plus-even'}
    tot = Counter(); c6 = Counter(); ends = Counter(); branch = Counter()
    for s_ in 'ABCD':

        hist = []

            hist.append((rule, T, P))

                    branch['U-branch (a singleton 1 or prefix met in mode U)'] += 1

                    ends[end] += 1
            if rule != 'death' or s_ != 'A':

            # C4: the dance of an interior death on line 75

            good = N1 == 2 ** w - 1 and P1 == 0
            expect = ['odd'] * (2 ** w - 1) + ['hash-odd'] + ['odd'] * (2 ** w - 1) + ['bounce']
            got = [r for r, _, _ in hist[i + 1:i + 1 + len(expect)]]
            if len(got) < len(expect):

            tot['dances'] += 1
            # C6: follow the climb after the bounce to the event on a (index p - 1), compare with both branches
            qb = i + len(expect)                   # index of the bounce in hist
            a0 = fs[p - 1]
            st_, n, pred, end = climb_check(hist[qb][2], seqs[qb + 1:])
            if st_ != 'ok':
                c6['climb not completed (' + st_.split()[0] + ')'] += 1; continue
            hit = None
            for (hp, Tp, lab) in pred[:n]:

                    hit = (hp, Tp, lab); break
            if hit is None:
                c6['nested death inside the climb' if end == 'death-in-stage' else 'event on a not reached'] += 1
                continue
            hp, Tp, lab = hit
            fs2, ss2 = Tp[0::2], Tp[1::2]
            lo2 = region_of(fs2, ss2, p - 1)
            B2, N2, P2 = region_state(fs2[lo2:-1], fs2[-1])

            if lab == 'plus':          # D-end: a - 3, a's bit flipped, direction reversed

            elif lab == 'odd-plus':    # U-end: a - 2, a's bit kept, direction kept

            else:
                c6['other end event ' + lab] += 1; continue

            c6[('D-end (reversed)' if lab == 'plus' else 'U-end (kept)')] += 1
            if not good:
                ok = False; print('    C6 fails at', fmt(T0)[:100], lab, B2, exp_bits, P2, exp_P)
    # random climbs, singletons of value 1 included, and the reviewer's reproducer

        if any(lab == 'plus-merge-1' for _, _, lab in pred[:n]):
            rnd_res['with PLUS-MERGE-1'] += 1
        if any(lab == 'death-merge' for _, _, lab in pred[:n]):

    ok &= rnd_res['mismatch'] == 0
    T = [9, PLUS, 5, PLUS, 1, PLUS, 7, PLUS, 5, PIPE, 1]

    print(f'    re-pairings on the four orbits to {far:.0e}: {dict(tot)}; {dict(branch)}; climb ends {dict(ends)}')
    print(f'    C6 on line 75 deaths whose dance and climb complete: {dict(c6)}')
    print(f'    random climbs: {dict(rnd_res)}')
    report('H', ok, 'C4 dance, C5 climb automaton (both modes, stage by stage, exact lists) and C6 (D-end reversed, '
           'U-end kept) on the orbits; random climbs with singletons 1')


# ---------------------------------------------------------------- I. fast exact map (counter-lemma jumps)
def jump(T):
    """Apply the maximal run of ODD half-cycles of the active region that hits no zero field (C1/C2).
    Returns (T', half-cycles, exact machine steps)."""
    fs, ss = list(T[0::2]), list(T[1::2]); k = len(fs) - 1
    lo = region_of(fs, ss, k)
    n = k - lo
    if n <= 0:
        return T, 0, 0
    Bv, N, P = region_state(fs[lo:k], fs[k])
    run = (1 << n) - 1 - N if P == 1 else N
    if run <= 0:
        return T, 0, 0

    def kth(u, kk):
        m = 1 << (u + 1); hh = 1 << u
        if P == 1:
            x0 = N + 1 + ((hh - (N + 1)) % m)
            return x0 + (kk - 1) * m - N
        x0 = N - ((N - hh) % m)
        return N - (x0 - (kk - 1) * m) + 1

    def cnt(u, A, B_):
        return (B_ >> u) - (B_ >> (u + 1)) - ((A >> u) - (A >> (u + 1)))
    s = run
    for i in range(lo, k):
        tt = kth(k - 1 - i, fs[i] + 1)
        if tt <= s:
            s = tt - 1
    if s <= 0:
        return T, 0, 0
    A, Bv_ = (N, N + s) if P == 1 else (N - s, N)
    steps = 6 * (s * fs[k] + 3 * s * (s - 1) // 2)
    cv = [cnt(v, A, Bv_) for v in range(n)]
    steps += sum(c * (12 * (v + 1) + 12) for v, c in enumerate(cv))
    for u in range(n):
        fu = fs[k - 1 - u]; M = 1 << (u + 1)
        cm = (Bv_ // M) - (A // M)
        if cm > 0:
            c0 = cnt(u, A, (A // M + 1) * M - 1) if P == 1 else cnt(u, (Bv_ // M) * M, Bv_)
            steps += 6 * (cm * fu - (cm * c0 + cm * (cm - 1) // 2))
    for v in range(n):
        fs[k - 1 - v] -= cv[v]
    fs[k] += 3 * s
    return mk(fs, ss), s, steps


def fast_run(T, max_steps, log=None):
    t = 0; h = 0
    while t < max_steps:
        T2, s, st = jump(T)
        if s:
            T = T2; h += s; t += st
            if t >= max_steps:
                break
        P, st, rule = step(T)
        if P is None:
            return T, h, t, rule
        if log is not None and rule != 'odd':
            log.append((h, t, rule))
        T = P; h += 1; t += st
    return T, h, t, None
