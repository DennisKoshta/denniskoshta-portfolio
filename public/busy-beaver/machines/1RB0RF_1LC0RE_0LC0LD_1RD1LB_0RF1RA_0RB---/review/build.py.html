# ============================================================== 1. the abstract relation R (proof.md section 2)
DIGC = {'L': '110', '0': '000', '1': '100'}


def inc(r):
    """+1 on the little-endian binary number of the 0/1 digits of r, L skipped; carry out appends a 1."""
    r = list(r)
    for i, d in enumerate(r):
        if d == 'L':
            continue
        if d == '0':
            r[i] = '1'
            return ''.join(r)
        r[i] = '0'
    return ''.join(r) + '1'


def cstep(tp, k, rest, ours=True, z=True):
    """One clause of R. Returns dict or None (no clause applies)."""
    if tp == 'hx':
        if not rest:
            return None
        d, r = rest[0], rest[1:]
        if d == 'L':
            return dict(cl='X1', rd='L', out='L', tp='hx', k=k + 1, rest=inc(r) if ours else r, inc=ours, done=False)
        if d == '0':
            return dict(cl='X2', rd='0', out='L', tp='hx', k=k + 1, rest=r, inc=False, done=False)
        return dict(cl='X3', rd='1', out='L', tp='h1', k=k, rest=r, inc=False, done=False)
    if not rest:
        if k == 0:
            return dict(cl='R0', rd='end', out='1', tp='h1', k=0, rest='', inc=False, done=True)
        if k % 2:
            return dict(cl='R1', rd='end', out='0', tp='h1', k=(k - 1) // 2, rest='', inc=False, done=False)
        return dict(cl='R2', rd='end', out='1', tp='h1', k=(k - 2) // 2, rest='', inc=False, done=False)
    if k == 0:
        if z and rest == '1':
            return dict(cl='Z1', rd='1', out='L', tp='h1', k=0, rest='', inc=False, done=True)
        if z and rest == '0':
            return dict(cl='Z0', rd='0', out='1', tp='h1', k=0, rest='', inc=False, done=True)
        return None
    d, r = rest[0], rest[1:]
    if d == 'L':
        return dict(cl='C1', rd='L', out='L', tp='hx', k=k, rest=inc(r) if ours else r, inc=ours, done=False)
    if d == '1':
        return dict(cl='C2', rd='1', out='L', tp='h1', k=k - 1, rest=r, inc=False, done=False)
    if k % 2:
        return dict(cl='C3', rd='0', out='0', tp='h1', k=(k - 1) // 2, rest=r, inc=False, done=False)
    return dict(cl='C4', rd='0', out='1', tp='h1', k=(k - 2) // 2, rest=r, inc=False, done=False)


def derive(tp, k, r, ours=True, z=True):
    """Full derivation of R(tp,k,r,_). Returns (steps, output or None)."""
    steps, out = [], ''
    while True:
        s = cstep(tp, k, r, ours, z)
        if s is None:
            steps.append(dict(cl='stuck', tp=tp, k=k, rest=r))
            return steps, None
        s['tp0'], s['k0'], s['rest0'] = tp, k, r
        steps.append(s)
        out += s['out']
        if s['done']:
            return steps, out
        tp, k, r = s['tp'], s['k'], s['rest']


def sweep(r, **kw):
    return derive('hx', 0, r, **kw)[1]


def Inv(tp, k, j, tq, q):
    if tp == 'hx' and tq == 'hx':
        return k <= q and j <= q
    if tp == 'h1' and tq == 'hx':
        return k + 1 <= q and j <= q
    if tp == 'h1' and tq == 'h1':
        return (j == 0 and 2 * k + 1 <= q) or (1 <= j and k + j <= q and 1 <= q)
    return False


def joint(r0):
    """The key lemma's induction, run on one word: first sweep over r0, second over its output."""
    r1 = sweep(r0)
    assert r1 is not None
    tp, k, rest, tq, q, srest, j = 'hx', 0, r0, 'hx', 0, r1, 0
    rows = []
    while True:
        K = k + j
        assert derive(tp, K, rest)[1] == srest, 'Lemma A link broken'
        assert Inv(tp, k, j, tq, q), 'Inv fails'
        if tp == 'h1' and (rest == '' or (k == 0 and rest in ('0', '1'))):
            s1 = cstep(tp, k, rest)
            fin_steps, fin = derive(tq, q, srest)
            assert fin is not None
            rows.append(dict(end=True, cl=s1['cl'], tp=tp, k=k, j=j, K=K, tq=tq, q=q, srest=srest,
                             fin=fin, fin_cls=[s['cl'] for s in fin_steps]))
            return r1, rows
        s1 = cstep(tp, k, rest)
        s2 = cstep(tq, q, srest)
        assert s1 and s2 and not s1['done'] and not s2['done']
        if s1['cl'] in ('C3', 'C4'):
            A = (K - 1) // 2 if K % 2 else (K - 2) // 2
            j2 = A - s1['k']
        else:
            j2 = j + 1
        rows.append(dict(end=False, n=len(rows) + 1, c1=s1['cl'], tp=tp, k=k, d1=rest[0], o1=s1['out'],
                         c2=s2['cl'], tq=tq, q=q, d2=srest[0], j=j, K=K))
        tp, k, rest, tq, q, srest, j = s1['tp'], s1['k'], s1['rest'], s2['tp'], s2['k'], s2['rest'], j2


# ============================================================== 2. the real machine
def parse_tm(s):
    t = {}
    for i, blk in enumerate(s.split('_')):
        for b in (0, 1):
            x = blk[3 * b:3 * b + 3]
            t[('ABCDEF'[i], b)] = None if x[2] == '-' else (int(x[0]), 1 if x[1] == 'R' else -1, x[2])
    return t


def S_word(tape, pos, st):
    """If the config is 0^inf 11 D> <r> 0^inf, return r (trailing 0 digits dropped), else None."""
    if st != 'D' or tape.get(pos - 1, 0) != 1 or tape.get(pos - 2, 0) != 1:
        return None
    if any(v for p, v in tape.items() if p < pos - 2):
        return None
    mx = max([p for p, v in tape.items() if v] + [pos])
    cells = [tape.get(p, 0) for p in range(pos, mx + 1)]
    while len(cells) % 3:
        cells.append(0)
    w = ''
    for i in range(0, len(cells), 3):
        c = ''.join(map(str, cells[i:i + 3]))
        if c not in ('110', '000', '100'):
            return None
        w += {'110': 'L', '000': '0', '100': '1'}[c]
    return w.rstrip('0')


def S_tape(r, pos=0):
    tape = {pos - 2: 1, pos - 1: 1}
    for i, d in enumerate(r):
        for j, c in enumerate(DIGC[d]):
            tape[pos + 3 * i + j] = int(c)
    return tape


def real_sweep(tms, r, cap=10 ** 6):
    """From S(r) with the head at cell 0, run to the first S config one cell further left."""
    t = parse_tm(tms)
    tape, pos, st = S_tape(r), 0, 'D'
    for n in range(1, cap):
        w, mv, nx = t[(st, tape.get(pos, 0))]
        tape[pos] = w
        pos += mv
        st = nx
        if pos == -1:
            s = S_word(tape, pos, st)
            if s is not None:
                return s, n
    raise RuntimeError('no boundary')


def run_rows(tms, tape, pos, st, nsteps=None, stop=None):
    """Simulate; return list of (step, state, head, tape copy, rule fired to get here)."""
    t = parse_tm(tms)
    rows = [(0, st, pos, dict(tape), None)]
    n = 0
    while True:
        if nsteps is not None and n >= nsteps:
            break
        sym = tape.get(pos, 0)
        rule = f'{st}{sym}'
        w, mv, nx = t[(st, sym)]
        tape[pos] = w
        pos += mv
        st = nx
        n += 1
        rows.append((n, st, pos, dict(tape), rule))
        if stop and stop(pos, st):
            break
    return rows

def md_inline(s):
    codes = []

    def keep(m):
        codes.append(m.group(1))
        return f'\x00{len(codes) - 1}\x00'
    s = re.sub(r'`([^`]+)`', keep, s)
    s = esc(s)
    s = re.sub(r'\[([^\]]+)\]\(([^)\s]+)\)', lambda m: f'<a href="/busy-beaver/machines/1RB0RF_1LC0RE_0LC0LD_1RD1LB_0RF1RA_0RB---/review/</span><span class="si">{m.group(2)}">{m.group(1)}</a>', s)
    s = re.sub(r'(?<![="])(https?://[^\s<)]+)', r'<a href="/busy-beaver/machines/1RB0RF_1LC0RE_0LC0LD_1RD1LB_0RF1RA_0RB---/review/\1">\1</a>', s)
    s = re.sub(r'\*\*(.+?)\*\*', r'<strong>\1</strong>', s)
    s = re.sub(r'(?<![\w*])\*(?![\s*])(.+?)(?<![\s*])\*(?![\w*])', r'<em>\1</em>', s)
    s = re.sub('\x00(\\d+)\x00', lambda m: '<code>' + esc(codes[int(m.group(1))]) + '</code>', s)
    return s


def md_slug(text, used):
    m = re.match(r'^(\d+(?:\.\d+)*)\.?\s', text)
    base = 's' + m.group(1).replace('.', '-') if m else re.sub(r'[^a-z0-9]+', '-', text.lower()).strip('-') or 'h'
    sid, n = base, 2
    while sid in used:
        sid, n = f'{base}-{n}', n + 1
    used.add(sid)
    return sid


def md_cells(line):
    line = line.strip()
    if line.startswith('|'):
        line = line[1:]
    if line.endswith('|'):
        line = line[:-1]
    cells, cur, inb = [], '', False
    for ch in line:
        if ch == '`':
            inb = not inb
        if ch == '|' and not inb:
            cells.append(cur)
            cur = ''
        else:
            cur += ch
    cells.append(cur)
    return [c.strip() for c in cells]


def md(text):
    """Return (html, [(level, id, text)])."""
    lines = text.split('\n')
    out, toc, used = [], [], set()
    stack = []          # [indent, tag, li_has_text]
    para = []
    para_in_li_new = [False]

    def flush():
        if not para:
            return
        txt = md_inline(' '.join(x.strip() for x in para))
        if stack and not para_in_li_new[0]:
            out.append(txt)
            stack[-1][2] = True
        else:
            out.append(f'<p>{txt}</p>')
        para.clear()
        para_in_li_new[0] = False

    def close_to(indent):
        while stack and stack[-1][0] > indent:
            out.append(f'</li></{stack[-1][1]}>')
            stack.pop()

    i, blank = 0, False
    while i < len(lines):
        ln = lines[i]
        s = ln.strip()
        ind = len(ln) - len(ln.lstrip(' '))
        if not s:
            flush()
            blank = True
            i += 1
            continue
        if s.startswith('```'):
            flush()
            if ind == 0:
                close_to(-1)
            j = i + 1
            buf = []
            while j < len(lines) and not lines[j].strip().startswith('```'):
                buf.append(lines[j][ind:] if lines[j][:ind].strip() == '' else lines[j])
                j += 1
            out.append('<pre class="md">' + esc('\n'.join(buf)) + '</pre>')
            i, blank = j + 1, False
            continue
        m = re.match(r'^(#{1,6})\s+(.*)$', s)
        if m and ind == 0:
            flush()
            close_to(-1)
            lvl, t = len(m.group(1)), m.group(2)
            sid = md_slug(re.sub(r'[`*]', '', t), used)
            toc.append((lvl, sid, t))
            out.append(f'<h{lvl} id="{sid}">{md_inline(t)} <a class="hl" href="/busy-beaver/machines/1RB0RF_1LC0RE_0LC0LD_1RD1LB_0RF1RA_0RB---/review/build.py.html#</span><span class="si">{sid}">#</a></h{lvl}>')
            i, blank = i + 1, False
            continue
        if s.startswith('|') and ind == 0:
            flush()
            close_to(-1)
            rows = []
            while i < len(lines) and lines[i].strip().startswith('|'):
                rows.append(lines[i])
                i += 1
            head = md_cells(rows[0])
            align = []
            body = rows[1:]
            if body and re.match(r'^\|?\s*:?-{3,}', body[0].strip()):
                align = ['r' if c.endswith(':') and not c.startswith(':') else '' for c in md_cells(body[0])]
                body = body[1:]
            t = ['<div class="tscroll"><table class="md"><thead><tr>']
            for n, c in enumerate(head):
                a = ' class="r"' if n < len(align) and align[n] == 'r' else ''
                t.append(f'<th{a}>{md_inline(c)}</th>')
            t.append('</tr></thead><tbody>')
            for r in body:
                t.append('<tr>')
                for n, c in enumerate(md_cells(r)):
                    a = ' class="r"' if n < len(align) and align[n] == 'r' else ''
                    t.append(f'<td{a}>{md_inline(c)}</td>')
                t.append('</tr>')
            t.append('</tbody></table></div>')
            out.append(''.join(t))
            blank = False
            continue
        if s.startswith('>') and ind == 0:
            flush()
            close_to(-1)
            buf = []
            while i < len(lines) and lines[i].strip().startswith('>'):
                buf.append(lines[i].strip()[1:].strip())
                i += 1
            out.append('<blockquote>' + md_inline(' '.join(buf)) + '</blockquote>')
            blank = False
            continue
        m = re.match(r'^(\s*)([-*]|\d+\.)\s+(.*)$', ln)
        if m:
            flush()
            tag = 'ol' if m.group(2)[0].isdigit() else 'ul'
            close_to(ind)
            if stack and stack[-1][0] == ind:
                if stack[-1][1] == tag:
                    out.append('</li>')
                else:
                    out.append(f'</li></{stack[-1][1]}>')
                    stack.pop()
                    out.append(f'<{tag}>')
                    stack.append([ind, tag, False])
            else:
                out.append(f'<{tag}>')
                stack.append([ind, tag, False])
            out.append('<li>')
            stack[-1][2] = False
            para.append(m.group(3))
            para_in_li_new[0] = False
            i, blank = i + 1, False
            continue
        # plain text
        if stack and ind > stack[-1][0]:
            if blank:
                flush()
                para_in_li_new[0] = True
            para.append(s)
        else:
            if stack and ind <= stack[0][0] and (blank or ind == 0):
                flush()
                close_to(-1)
            para.append(s)
        i, blank = i + 1, False
    flush()
    close_to(-1)
    return '\n'.join(out), toc

def coq_tokens(src):
    """Yield (cls, text); cls in kw/tac/com/str/num/bul/None."""
    i, n = 0, len(src)
    bol = True
    while i < n:
        if src.startswith('(*', i):
            depth, j = 1, i + 2
            while j < n and depth:
                if src.startswith('(*', j):
                    depth += 1
                    j += 2
                elif src.startswith('*)', j):
                    depth -= 1
                    j += 2
                else:
                    j += 1
            yield 'com', src[i:j]
            i, bol = j, False
            continue
        c = src[i]
        if c == '"':
            j = src.index('"', i + 1) + 1
            yield 'str', src[i:j]
            i, bol = j, False
            continue
        if c == '\n':
            yield None, c
            i, bol = i + 1, True
            continue
        if c == ' ':
            j = i
            while j < n and src[j] == ' ':
                j += 1
            yield None, src[i:j]
            i = j
            continue
        if bol and c in '-+*' and i + 1 < n and src[i + 1] == ' ':
            yield 'bul', c
            i, bol = i + 1, False
            continue
        m = re.compile(r"[A-Za-z_][A-Za-z0-9_']*").match(src, i)
        if m:
            w = m.group(0)
            yield ('kw' if w in COQ_KW else 'tac' if w in COQ_TAC else None), w
            i, bol = m.end(), False
            continue
        m = re.compile(r'\d+').match(src, i)
        if m:
            yield 'num', m.group(0)
            i, bol = m.end(), False
            continue
        yield None, c
        i, bol = i + 1, False


def coq_lines(src):
    """List of per-line HTML strings (spans closed at each line end)."""
    lines, cur = [], []
    for cls, text in coq_tokens(src):
        parts = text.split('\n')
        for pi, p in enumerate(parts):
            if pi > 0:
                lines.append(''.join(cur))
                cur = []
            if p:
                cur.append(f'<span class="{cls}">{esc(p)}</span>' if cls else esc(p))
    lines.append(''.join(cur))
    return lines

def find(lines, pat, lo=1, hi=None, nth=1):
    """1-based line number of the nth line in [lo,hi] matching regex pat."""
    hi = hi or len(lines)
    c = 0
    for n in range(lo, hi + 1):
        if re.search(pat, lines[n - 1]):
            c += 1
            if c == nth:
                return n
    raise KeyError(pat)

def word(w, before=None):
    """Digit chips; with before, outline digits that differ."""
    out = []
    for i, d in enumerate(w):
        ch = before is not None and (i >= len(before) or before[i] != d)
        out.append(dg(d, ch))
    if not w:
        out.append('<span class="dg end">[]</span>')
    return '<span class="word">' + ''.join(out) + '</span>'


def pw(w):
    """Word with a leading L-run as a power: LLLL1L1 -> L⁴ 1L1."""
    sup = str.maketrans('0123456789', '⁰¹²³⁴⁵⁶⁷⁸⁹')
    n = len(w) - len(w.lstrip('L'))
    if n >= 2:
        return f'L{str(n).translate(sup)} {w[n:]}'.strip()
    return w


def tape_cells(cells, head=None, state=None, groups=(), left_inf=True, right_inf=True, head_label=None, marks=None):
    """cells: list of (absolute index, value). groups: (first idx, last idx, label, kind). marks: {idx: text} above cells."""
    PITCH = 21                       # .cell is 22 px wide with a -1 px overlap
    cols = []
    if left_inf:
        cols.append('<div class="cellcol"><div class="hmark"></div><div class="dots">0<sup>∞</sup></div></div>')
    for (p, v) in cells:
        hm = (head_label or (state + '▸' if state else '▼')) if p == head else (marks or {}).get(p, '')
        mk = ' mk' if (marks and p in marks and p != head) else ''
        cols.append(f'<div class="cellcol"><div class="hmark{mk}">{esc(hm)}</div>'
                    f'<div class="cell c{v}{" hd" if p == head else ""}" title="cell {p}: {v}">{v}</div></div>')
    if right_inf:
        cols.append('<div class="cellcol"><div class="hmark"></div><div class="dots">0<sup>∞</sup></div></div>')
    x0 = cells[0][0]
    off = 26 if left_inf else 0
    lab = []
    for (a, b, label, kind) in groups:
        x, w = (a - x0) * PITCH + off, (b - a + 1) * PITCH
        lab.append(f'<div class="gl {kind}" style="position:absolute;left:{x + 1}px;width:{w - 3}px">{esc(label)}</div>')
    total = off + len(cells) * PITCH + (26 if right_inf else 0)
    return (f'<div style="display:inline-block;vertical-align:top"><div class="tape">{"".join(cols)}</div>'
            f'<div class="glrow" style="width:{total}px">{"".join(lab)}</div></div>')

def spacetime(rows, x0, x1, cell=15, top=(), bottom=(), rule_col=True):
    """SVG space-time diagram, one row per step. top/bottom: digit brackets (first, last, label) above/below."""
    lw = 96
    W = lw + (x1 - x0 + 1) * cell + 8
    ytop = 16 + (18 if top else 0)
    H = ytop + len(rows) * cell + (22 if bottom else 6)
    o = [f'<svg class="st" width="{W}" height="{H}" viewBox="0 0 {W} {H}" role="img">']
    for p in range(x0, x1 + 1):
        if p % 3 == 0:
            o.append(f'<text class="lab" x="{lw + (p - x0) * cell + cell / 2}" y="11" text-anchor="middle">{p}</text>')

    def brackets(gs, y, ty):
        for (a, b, lab) in gs:
            xa, xb = lw + (a - x0) * cell + 1, lw + (b - x0 + 1) * cell - 1
            o.append(f'<line class="gl" x1="{xa}" y1="{y}" x2="{xb}" y2="{y}"/>'
                     f'<text class="gt" x="{(xa + xb) / 2}" y="{ty}" text-anchor="middle">{esc(lab)}</text>')
    brackets(top, ytop - 3, ytop - 6)
    for i, (n, st, pos, tape, rule) in enumerate(rows):
        y = ytop + i * cell
        txt = f'{n:>3} {st}' + (f' ({rule})' if rule and rule_col else '')
        o.append(f'<text class="lab" x="2" y="{y + cell - 4}">{esc(txt)}</text>')
        for p in range(x0, x1 + 1):
            v = tape.get(p, 0)
            o.append(f'<rect class="c{v}" x="{lw + (p - x0) * cell}" y="{y}" width="{cell}" height="{cell}"/>')
        if x0 <= pos <= x1:
            o.append(f'<rect class="hd" x="{lw + (pos - x0) * cell + 1.2}" y="{y + 1.2}" width="{cell - 2.4}" height="{cell - 2.4}"/>')
    yb = ytop + len(rows) * cell + 3
    brackets(bottom, yb, yb + 13)
    o.append('</svg>')
    return ''.join(o)

def mins(t):
    p = [int(x) for x in t.split(':')]
    return p[0] * 60 + p[1] + (p[2] / 60 if len(p) > 2 else 0)


def gantt(lo, hi, rows, title, ppm=5.0, extra=180):
    lw = 210
    W = int(lw + (hi - lo) * ppm + extra)
    rh = 22
    H = 40 + len(rows) * rh + 8
    o = [f'<svg class="gantt" width="{W}" height="{H}" viewBox="0 0 {W} {H}" role="img" aria-label="{esc(title)}">',
         f'<text class="ttl" x="0" y="13">{esc(title)}</text>']
    t = lo - lo % 30 + (30 if lo % 30 else 0)
    while t <= hi:
        x = lw + (t - lo) * ppm
        o.append(f'<line class="grid" x1="{x}" y1="34" x2="{x}" y2="{H - 6}"/>'
                 f'<text class="tick" x="{x}" y="31" text-anchor="middle">{t // 60:02d}:{t % 60:02d}</text>')
        t += 30
    for i, (lab, items) in enumerate(rows):
        y = 40 + i * rh
        o.append(f'<text class="lab" x="0" y="{y + 14}">{esc(lab)}</text>')
        for (s, e, kind, tip) in items:
            x = lw + (mins(s) - lo) * ppm
            if kind == 'a':
                w = max(3, (mins(e) - mins(s)) * ppm)
                o.append(f'<rect class="b1" x="{x:.1f}" y="{y + 3}" width="{w:.1f}" height="14" rx="3"><title>{esc(tip)}</title></rect>')
                d = int(round(mins(e) - mins(s)))
                o.append(f'<text class="dur" x="{x + w + 4:.1f}" y="{y + 14}">{s}–{e} ({d} min)</text>')
            elif kind == 'd':
                o.append(f'<path class="b2" d="M{x:.1f},{y + 2} l7,8 l-7,8 l-7,-8 z"><title>{esc(tip)}</title></path>'
                         f'<text class="dur" x="{x + 10:.1f}" y="{y + 14}">{esc(tip)}</text>')
            else:
                o.append(f'<circle class="ok" cx="{x:.1f}" cy="{y + 10}" r="3.5"><title>{esc(s)} {esc(tip)}</title></circle>')
        if items and items[0][2] == 'p':
            o.append(f'<text class="dur" x="{lw + (mins(items[-1][0]) - lo) * ppm + 8:.1f}" y="{y + 14}">'
                     f'{" · ".join(s[:5] + " " + tip.split(" (")[0] for (s, _, _, tip) in items)}</text>')
    o.append('</svg>')
    return ''.join(o)
