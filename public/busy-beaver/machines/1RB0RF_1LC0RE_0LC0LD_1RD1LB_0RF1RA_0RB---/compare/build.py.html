"""compare/build.py -- builds ../compare.html (#2 vs SOMDCv6 TM7) plus compare/TM7.html and compare/stats.json.

Every number on the page is computed here from the files:
  ours  = machines/<tm>/<tm>.v
  TM7   = vendor/busycoq/verify/SOMDCv6.v: the shared header (lines before `Module TM1.`) + the body of
          `Module TM7.` (the `Module TM7.` / `End TM7.` lines themselves are the "module wrapper", left out).
          rocq-scratch/TM7_toplevel.v is checked to be byte-identical to that, and the result is printed on the page.
Line diff: difflib.ndiff on trailing-space-trimmed lines, blank lines as junk, the TM string replaced by <TM>
in both files. ndiff pairs a removed and an added line as "changed" when they are similar (its 0.75 cutoff).
Counts are over non-blank lines.

Run:  [local working file] -B compare/build.py      (needs pygments)
"""

ROOT = Path("[local working file]")
MD = ROOT / "machines" / TM
OURS = MD / f"{TM}.v"
VER = ROOT / "vendor/busycoq/verify"
SOMD = VER / "SOMDCv6.v"
TOP = MD / "rocq-scratch/TM7_toplevel.v"

def sha(p):

# ---------------------------------------------------------------- sources

def norm(t):
    return TMSTR_RE.sub('TM_from_str "<TM>"', t.rstrip())


# ---------------------------------------------------------------- syntax classes per char
LEX = get_lexer_by_name("rocq", stripnl=False, stripall=False, ensurenl=True)


def tok_class(tt):
    if tt in Token.Comment:
        return "cm"
    if tt in Token.Keyword.Namespace:
        return "kv"
    if tt in Token.Keyword:
        return "kt"
    if tt in Token.Literal.String:
        return "st"
    if tt in Token.Literal.Number:
        return "nu"
    if tt in Token.Name.Builtin:
        return "nb"
    if tt in Token.Operator or tt in Token.Punctuation:
        return "op"
    return ""


def lex_lines(lines):
    text = "\n".join(lines) + "\n"
    chars = []
    for tt, v in LEX.get_tokens(text):
        c = tok_class(tt)
        chars.extend([c] * len(v))
    out, pos = [], 0
    for l in lines:
        out.append(chars[pos:pos + len(l)])
        pos += len(l) + 1
    rebuilt = "".join(v for _, v in LEX.get_tokens(text))
    assert rebuilt == text, "lexer changed the text"
    return out


tm7_show = [(i + 1, somd[i]) for i in hdr_idx] + [(i + 1, somd[i]) for i in range(t7s, t7e + 1)]

def render(text, cls, flags=None):
    """text + per-char syntax classes (+ per-char diff flags) -> html"""
    if not text:
        return ""
    flags = flags or [""] * len(text)
    out, i = [], 0
    while i < len(text):
        j = i
        while j < len(text) and cls[j] == cls[i] and flags[j] == flags[i]:
            j += 1
        c = " ".join(x for x in (cls[i], flags[i]) if x)
        seg = E(text[i:j])
        out.append(f'<span class="{c}">{seg}</span>' if c else seg)
        i = j
    return "".join(out)


TOKEN_RE = re.compile(r"[A-Za-z0-9_']+|\s+|[^\sA-Za-z0-9_']")


TMSTR_TOK = re.compile(r"((?:[0-9\-][RL\-][A-Z\-]){2}(?:_(?:[0-9\-][RL\-][A-Z\-]){2})+)")


def tokens(s):
    """word / space / punctuation tokens; a machine string is split into its 3-char transitions."""
    out = []
    for n, part in enumerate(TMSTR_TOK.split(s)):
        if n % 2:
            out.extend(re.findall(r"_|.{3}", part))
        elif part:
            out.extend(TOKEN_RE.findall(part))
    assert "".join(out) == s
    return out


def intraline(a, b):
    """per-char flags for a changed line pair (token-level)."""
    ta, tb = tokens(a), tokens(b)
    fa, fb = [""] * len(a), [""] * len(b)
    pa = [0]
    for t in ta:
        pa.append(pa[-1] + len(t))
    pb = [0]
    for t in tb:
        pb.append(pb[-1] + len(t))
    for op, i1, i2, j1, j2 in difflib.SequenceMatcher(None, ta, tb, autojunk=False).get_opcodes():
