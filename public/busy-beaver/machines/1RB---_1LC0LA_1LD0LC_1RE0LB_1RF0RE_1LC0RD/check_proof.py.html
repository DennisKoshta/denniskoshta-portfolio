def table(tm):
    return rlsim.parse(tm)


def maps_between(src, dst):
    """All (renaming pi, mirror) with dst[pi q][s] = (w, +-d, pi q') whenever src[q][s] = (w, d, q')."""
    A, B = table(src), table(dst); out = []
    for perm in itertools.permutations('ABCDEF'):
        pi = dict(zip('ABCDEF', perm))
        for mirror in (False, True):
            ok = True
            for (q, s), t in A.items():
                u = B[(pi[q], s)]
                if t is None or u is None:
                    ok &= t is None and u is None
                else:
                    ok &= u == (t[0], -t[1] if mirror else t[1], pi[t[2]])
                if not ok: break
            if ok: out.append((''.join(perm), mirror))
    return out


def lockstep(n):
    """This machine from the blank tape against the parent's table from 0^inf F> 0^inf, transition by transition:
    same symbol read and written, opposite move, state = sigma(parent state); cell x here is cell -x there."""
    A, P = table(TM), table(PARENT)
    ta, tp = {}, {}; qa, qp = 'A', 'F'; pa = pp = 0
    for k in range(n):
        sa, sp = ta.get(pa, 0), tp.get(pp, 0)
        x, y = A[(qa, sa)], P[(qp, sp)]
        if x is None or y is None or sa != sp or qa != SIGMA[qp] or x != (y[0], -y[1], SIGMA[y[2]]):
            return False, k
        ta[pa] = x[0]; tp[pp] = y[0]; pa += x[1]; pp += y[1]; qa, qp = x[2], y[2]
    return {-x: v for x, v in ta.items()} == tp and pa == -pp and qa == SIGMA[qp], n


def lam_direct(tm, start, n, mirror):
    """Direct simulation: (step, cell) at every change of the leftmost cell that has held 1, in the frame where cell x
    is cell -x of the machine's tape if mirror (so it tracks the rightmost such cell of the machine's tape)."""
    T = table(tm); tape = {}; q = start; p = 0; lam = None; log = []
    sgn = -1 if mirror else 1
    for k in range(1, n + 1):
        w, mv, nq = T[(q, tape.get(p, 0))]
        tape[p] = w
        if w == 1 and (lam is None or sgn * p < lam): lam = sgn * p; log.append((k, lam))
        p += mv; q = nq
    return log

def defect_state(lo_c, cells, h):
    """Defect description of a tape with the head on cell h moving right (proof.md 2.1 of the parent): lo, hi, the
    pre-sweep defects strictly between them (edge h - 1 toggled), head cell, phase phi (B1 place: h = phi + 3)."""
    first1 = next(i for i, c in enumerate(cells) if c) + lo_c
    last1 = max(i for i, c in enumerate(cells) if c) + lo_c
    rd = lambda x: cells[x - lo_c] if 0 <= x - lo_c < len(cells) else 0
    D = set(e for e in range(first1, last1) if rd(e) == rd(e + 1))
    D ^= {h - 1}
    return dict(lo=first1 - 1, hi=last1 + 1, D=tuple(sorted(D)), p=h, dir=1, phi=(h - 3) % 4)


def snap_cb(store, rename):
    def cb(n, rl):
        lo_c, cells, h = rl.cells()

        st.update(steps=rl.steps, state=rename[rl.state], read=rl.head_symbol(), facing=rl.facing, lam=rl.lam,
                  own_state=rl.state)
        if n in (16, 28): st['runs'] = rl.runs(); st['cells'] = (lo_c, cells)
        store[n] = st
    return cb

def v_table():
    """V_r(u) as written in the parent's write-up (submission/proof.md, section 3.4)."""
    txt = open(os.path.join(PDIR, 'submission', 'proof.md')).read()
    sec = txt[txt.index('### 3.4 The window vectors'):txt.index('## 4. Three regions')]
    V = {}
    for line in sec.split('\n'):
        m = re.match(r'\| r([0-5]) \| (.*) \| (.*) \|$', line)
        if not m: continue
        def parse(cell):
            if cell.strip() == '(none)': return []
            return [re.sub(r'(\d+)u', r'\1*u', item.strip()) for item in cell.split(', ')]
        V[int(m.group(1))] = (parse(m.group(2)), parse(m.group(3)))
    return V


def ev_v(expr, u):
    val = eval(expr.replace('/', '*Fr(1)/'), {'u': u, 'Fr': Fr})
    assert Fr(val).denominator == 1
    return int(val)


LEFT_WORD = (-32, -27, -26, -21, -20, -15, -14, -9, -8, -4)
NLEFT = len(LEFT_WORD) + 1          # gaps lo -> first left-word defect -> ... -> A-4 -> A+2


def parse_C(s, n):
    """An independent reading of the parent's section 3.3 at record exit n: returns (ok, info). info has A, B, K, the
    chain gaps (in order), the right gaps (B -> x_1 -> ... -> hi), all gaps from lo to hi, and what failed."""
    r = cls(n); e = epoch(n); K = K0[r] + 2 * e
    D = list(s['D']); lo, hi = s['lo'], s['hi']; A = lo + 34; bad = []
    if s['lam'] != lo + 1: bad.append('lam != lo + 1')
    left = [x - A for x in D if x < A]
    if tuple(left) != LEFT_WORD: bad.append('(L)')
    rest = [x for x in D if x >= A]
    chain = rest[:4 * K - 2]; right = rest[4 * K - 2:]
    if len(chain) < 4 * K - 2 or chain[:2] != [A + 2, A + 3]: bad.append('block 1')
    gaps = [b - a for a, b in zip(chain, chain[1:])]
    roles = ['d'] + ['i', 'd', 's', 's'] * (K - 1)
    need = {'d': (1, 1), 's': (0, 4), 'i': (1, 5)}
    for g, ro in zip(gaps, roles):
        res, mn = need[ro]
        if g % 4 != res or g < mn: bad.append(f'gap {g} role {ro}')
    B = chain[-1] + 2
    if not (s['p'] == B - 1 and s['state'] == 'B' and s['read'] == 1 and s['facing'] > 0): bad.append('(H)')
    if right and right[0] <= B: bad.append('x_1 <= B')
    rgaps = [b - a for a, b in zip([B] + right, right + [hi])]
    allg = [b - a for a, b in zip([lo] + D, D + [hi])]
    return not bad, dict(A=A, B=B, K=K, lo=lo, hi=hi, gaps=gaps, roles=roles, right=rgaps, all=allg, bad=bad,
                         win=gaps[len(gaps) - J[r]:] if J[r] else [])


def blocks_of(info):
    """The chain by blocks: block 1 = [digit gap]; block j >= 2 = [inter, digit, sep, sep]."""
    g = info['gaps']
    return [[g[0]]] + [g[1 + 4 * j:5 + 4 * j] for j in range(info['K'] - 1)]

def parse_cfg28(vtext):
    m = re.search(r'Definition cfg28 : cfg :=\s*Rc (\w)\s*\(\[(.*?)\], T0\)\s*\(\[(.*?)\], T0\)\.', vtext, re.S)
    if not m: return None
    def runs(t): return [(1 if a == 'S1' else 0, int(b)) for a, b in re.findall(r'\((S[01]), cn (\d+)\)', t)]
    return m.group(1), runs(m.group(2)), runs(m.group(3))


def parse_nus(vtext):
    m = re.search(r'Definition nus28 : list \(list N\) :=\s*\[(.*?)\]%N\.', vtext, re.S)
    return [[int(x) for x in re.findall(r'\d+', blk)] for blk in re.findall(r'\[([\d; ]+)\]', m.group(1))]


def parse_L28(vtext):
    m = re.search(r'Definition L28 : list run :=\s*\[(.*?)\]%N\.', vtext, re.S)
    return [(1 if a == 'S1' else 0, int(b)) for a, b in re.findall(r'\((S[01]), cn (\d+)\)', m.group(1))]


def decode_nus(L):
    """nus from the left run list (nearest first): 6 blocks of 4 runs (t, s, a, lp) = (4+4x0, 4+4x1, 1+4x2, 5+4x3)."""
    out = []
    for j in range(6):
        r = [n for _, n in L[4 * j:4 * j + 4]]
        x = [r[0] - 4, r[1] - 4, r[2] - 1, r[3] - 5]
        if any(v % 4 or v < 0 for v in x): return None
        out.append([v // 4 for v in x])
    return out

def md_table(txt, start_marker):
    """The markdown table rows (as lists of cells) of the first table after start_marker."""
    if start_marker not in txt: return []
    i = txt.index(start_marker); out = []; seen = False
    for line in txt[i:].split('\n'):
        if line.startswith('|'):
            seen = True
            cells = [c.strip().replace('**', '') for c in line.strip().strip('|').split('|')]
            if not all(set(c) <= set('-:') for c in cells): out.append(cells)
        elif seen:
            break
    return out[1:]
