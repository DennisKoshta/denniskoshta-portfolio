# ---------------------------------------------------------------- TM simulator
def run(left, state, right, stop, maxs):
    """Tape = 0^inf left right 0^inf, head on right[0] in `state` (letter).
    stop(kind, pos, q) is called on 'C>' (C entered by a right move) and 'LREC' (new leftmost cell).
    Returns (status, steps, tape, head, q, base)."""
    pad = 4 * (len(left) + len(right)) + 64
    t = bytearray(pad) + bytearray(int(c) for c in left + right) + bytearray(pad)
    h = pad + len(left); base = h; q = ord(state) - 65; lo = pad; step = 0
    while step < maxs:
        if h <= 2 or h >= len(t) - 3:
            ext = bytearray(len(t)); t = ext + t + ext; h += len(ext); base += len(ext); lo += len(ext)
        tr = TAB[2 * q + t[h]]
        if tr is None:
            return 'HALT', step, t, h, q, base
        t[h] = tr[0]; h += tr[1]; q = tr[2]; step += 1
        if h < lo:
            lo = h
            if stop('LREC', h - base, q):
                return 'LREC', step, t, h, q, base
        elif q == 2 and tr[1] == 1 and stop('C>', h - base, q):
            return 'C>', step, t, h, q, base
    return 'MAX', step, t, h, q, base


def cells(t, a, b):

def Khub(a, b):

def carry_wrong(G):
    """WRONG reading of the S case (carry resumes after the toggled segment); used only as a mutation test."""
    G = list(G); i = 0
    while True:
        if i == len(G): G.append('001'); return G
        g = G[i]
        if g == '010': i += 1
        elif g == '001': G[i] = '000'; i += 1
        elif g == '000': G[i] = '001'; return G
        else:   # '100'
            G[i] = '010'; j = i + 1
            while True:
                if j == len(G): G.append('010'); break
                if G[j] == '010': G[j] = '000'
                elif G[j] == '000': G[j] = '010'; break
                j += 1
            i = j + 1


def value(x0, G):
    v = x0; k = 0
    for g in G:
        if g in ('000', '001'):

            if g == '001': v += 1 << k
    return v

# ------------------------------------------------- 2.5 L1 lemmas (proof.md 2.5), each checked as an exact window rule
def win(cl, pos, q, until=None, limit=10 ** 6):
    """Run the TM on the finite window cl (list of 0/1) from head pos in state q.
    Stops when the head leaves the window ('L'/'R') or until(h, q) is true ('STOP'). Returns (how, state, cells, steps, h)."""
    t = list(cl); h = pos; q = ord(q) - 65; s = 0
    while s < limit:
        if h < 0: return 'L', 'ABCDEF'[q], ''.join(map(str, t)), s, h
        if h >= len(t): return 'R', 'ABCDEF'[q], ''.join(map(str, t)), s, h
        if until and s > 0 and until(h, 'ABCDEF'[q]): return 'STOP', 'ABCDEF'[q], ''.join(map(str, t)), s, h
        tr = TR[(q, t[h])]
        if tr is None: return 'HALT', 'ABCDEF'[q], ''.join(map(str, t)), s, h
        t[h] = tr[0]; h += tr[1]; q = tr[2]; s += 1
    return 'MAX', 'ABCDEF'[q], ''.join(map(str, t)), s, h

def Cw(c):
    return sum(x << j for j, x in enumerate(c))


def layout_of(G):
    c = [0]
    for g in G:
        if g == '010': c[-1] += 1
        else: c.append(0)
    while len(c) > 1 and c[-1] == 0: c.pop()
    return c


def step_nc(n, c):
    """(n, c) at an eat -> (n', c', D, M).  D = normalization depth of the next residue."""
    c = list(c)
    if n & 1:
        H = c; n2 = n + 1; M = (n + 1) // 2 + Cw(c)
    else:
        V = n // 2 + Cw(c); t = 0
        while (V >> t) & 1: t += 1
        H = c[t + 1:]; n2 = n + 2 + t + sum(c[:t + 1]); M = V >> (t + 1)
    Js = [j for j, x in enumerate(H) if x > 0]
    D = 1 + sum(1 for j in range(max(Js)) if not (M >> j) & 1) if Js else 0
    c2 = [0]
    for j in range(max(len(H), M.bit_length()) + 1):
        c2[-1] += H[j] if j < len(H) else 0
        if (M >> j) & 1: c2[-1] += 1
        else: c2.append(0)
    while len(c2) > 1 and c2[-1] == 0: c2.pop()
    return n2, c2, D, M


def B(x):
    return x.bit_length()

def cascade_start(Bn, c):
    """odd n with M=(n+1)/2+C(c)=2^(B-1), B=B(n+1); shrinks c until C<=2^(B-2)-1. Returns (n, c) or None."""
    c = list(c)
    while Cw(c) > (1 << (Bn - 2)) - 1 and any(c):
        k = max(j for j, x in enumerate(c) if x > 0); c[k] -= 1
    n = (1 << Bn) - 2 * Cw(c) - 1
    return (n, c) if n > 0 and n % 2 == 1 and B(n + 1) == Bn else None


# 5.4 cascade lemma: (i) V'=2^B, M'=2^(B-1), D'=B-1, slack' = 0 if n''<2^B else 1;
#     (ii) c0 odd and n''+1<2^B: n'' odd, C''=3*2^(B-2)+(C-c0)/2 >= 2^(B(n''+1)-1) (Inv_S fails), 2M''=5*2^(B-1)+3-C, D''<=B-1
#     (iii) any c0: D''<=B-1, so the eat two epochs later is normalized
def cascade_check(n, c):
    Bn = B(n + 1); C = Cw(c)
    n1, c1, D0, M0 = step_nc(n, c)
    n2, c2, D1, M1 = step_nc(n1, c1)
    n3, c3, D2, M2 = step_nc(n2, c2)
    ok = (M0 == 1 << (Bn - 1) and n1 == n + 1 and n1 // 2 + Cw(c1) == 1 << Bn and M1 == 1 << (Bn - 1) and D1 == Bn - 1
          and B(n2) - 1 - D1 == (0 if n2 < (1 << Bn) else 1) and D2 <= Bn - 1 and B(n3) - 1 - D2 >= 0)
    odd = c[0] % 2 == 1 and n2 + 1 < (1 << Bn)
    if odd:
        ok = ok and n2 % 2 == 1 and Cw(c2) == 3 * (1 << (Bn - 2)) + (C - c[0]) // 2 \
             and Cw(c2) >= 1 << (B(n2 + 1) - 1) and 2 * M2 == 5 * (1 << (Bn - 1)) + 3 - C
    return ok, odd

def eat_word(n, c):
