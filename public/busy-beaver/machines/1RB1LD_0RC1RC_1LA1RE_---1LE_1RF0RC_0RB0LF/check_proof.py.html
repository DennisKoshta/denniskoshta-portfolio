def krun(jobs):
    """jobs: list of (state, headidx, nrec, maxsteps, tape) -> list of (kind, steps, state, headoff, word)"""
    if not jobs:
        return []
    inp = ''.join('%s %d %d %d %s\n' % j for j in jobs)
    out = subprocess.run([KRUN], input=inp, capture_output=True, text=True, check=True).stdout.split('\n')
    res = []
    for line in out:
        if not line:
            continue
        k, s, q, h, w = line.split()
        res.append((k, int(s), q, int(h), w))
    assert len(res) == len(jobs)
    return res

def run_window(cells, pos, st, nsteps):
    """run on a finite window; return (cells, pos, st, max_lo, max_hi) or None if the head reads outside"""
    cells = list(cells)
    for _ in range(nsteps):
        if not 0 <= pos < len(cells):
            return None
        tr = TABLE[(st, cells[pos])]
        if tr is None:
            return None
        cells[pos] = tr[0]
        pos += tr[1]
        st = tr[2]
    return cells, pos, st


def run_dict(tape, pos, st, nsteps):
    """unbounded reference run: tape = dict pos->bit"""
    for i in range(nsteps):
        tr = TABLE[(st, tape.get(pos, 0))]
        if tr is None:
            return ('HALT', i, tape, pos, st)
        tape[pos] = tr[0]
        pos += tr[1]
        st = tr[2]
    return ('OK', nsteps, tape, pos, st)

def parse_lin(s):
    """'n+3', 'a-2', '2b+5', 'c', '7n+14a+1135' -> {var: coef, '': const}"""
    s = s.replace(' ', '').replace('-', '+-')
    d = {'': 0}
    for term in s.split('+'):
        if not term:
            continue
        m = re.fullmatch(r'(-?\d*)([a-z]?)', term)
        assert m, term
        co, v = m.groups()
        if v:
            c = int(co) if co not in ('', '-') else (-1 if co == '-' else 1)
            d[v] = d.get(v, 0) + c
        else:
            d[''] += int(co)
    return d


def parse_word(s):
    """'1^(n+3) 0^3 1^2 0 1^c' -> tokens with exponents as linear forms in the ORIGINAL variables"""
    toks = []
    for t in s.split():
        m = re.fullmatch(r'([01])(?:\^(\d+|\([^)]*\)|[a-z]))?', t)
        assert m, t
        sym, ex = m.groups()
        e = {'': 1} if ex is None else (parse_lin(ex.strip('()')) if not ex.isdigit() else {'': int(ex)})
        toks.append((sym, e))
    return toks


def word_value(toks, val):

SHIFT = {

def check_SR():
    for name, (w0, h0, s0, n, w1, h1, s1) in SHIFT.items():
        cells = [int(c) for c in w0]
        # exact run that never reads outside the window (the last move may leave it: R1 ends just right of it)
        st, pos, ok = s0, h0, True
        for i in range(n):
            if not 0 <= pos < len(cells):
                ok = False
                break
            tr = TABLE[(st, cells[pos])]
            if tr is None:
                ok = False
                break
            cells[pos] = tr[0]
            pos += tr[1]
            st = tr[2]
        rec('SR.%s_window_run' % name, ok and ''.join(map(str, cells)) == w1 and pos == h1 and st == s1,
            (cells, pos, st))
        # k-fold form in random contexts, by unbounded simulation
        for k in range(0, 41):
            for _ in range(scale(2, 5, 20)):
                lc = [RNG.randrange(2) for _ in range(RNG.randrange(0, 8))]
                rc = [RNG.randrange(2) for _ in range(RNG.randrange(0, 8))]
                if name == 'R1':
                    word = lc + [1, 1] * k + rc
                    pos = len(lc)
                    exp = lc + [1, 0] * k + rc
                    epos = len(lc) + 2 * k
                    steps = 2 * k
                elif name == 'R2b':
                    word = lc + [0] + [1, 0] * k + [0, 0, 0] + rc
                    pos = len(lc) + 1 + 2 * k
                    exp = lc + [0] + [0, 0, 0] + [1, 1] * k + rc
                    epos = len(lc) + 1
                    steps = 12 * k
                else:
                    word = lc + [0] + [1, 0] * k + [1, 1, 0] + rc
                    pos = len(lc) + 1 + 2 * k + 2
                    exp = lc + [0] + [1, 1, 0] + [1, 1] * k + rc
                    epos = len(lc) + 3
                    steps = 12 * k
                tape = {i: b for i, b in enumerate(word)}
                kind, ns, tape, p, st = run_dict(tape, pos, s0, steps)
                got = [tape.get(i, 0) for i in range(min(tape) if tape else 0, max(len(word), max(tape) + 1))]
                ok = kind == 'OK' and st == s0 and p == epos and all(
                    tape.get(i, 0) == (exp[i] if 0 <= i < len(exp) else 0) for i in range(-3, len(word) + 3))
                rec('SR.%s_kfold_k<=40' % name, ok, (k, lc, rc, got))
        if name != 'R1':
            # teeth: with a 1 left of the crossed copies the 12-step result is different
            cells = [1] + [int(c) for c in w0[1:]]
            tape = {i: b for i, b in enumerate(cells)}
            kind, ns, tape, p, st = run_dict(tape, h0, s0, n)
            got = ''.join(str(tape.get(i, 0)) for i in range(len(cells)))
            rec('SR.%s_needs_the_0_on_the_left' % name, not (got == w1 and p == h1 and st == s1), (got, p, st))

def rand_tokens(rng, vars_):
    toks = []
    for _ in range(rng.randrange(3, 14)):
        r = rng.random()
        if r < 0.35:
            toks.append(('0',))
        elif r < 0.55:
            toks.append(('1', L(1)))
        elif r < 0.8:
            v = rng.choice(vars_)
            toks.append(('1', L(rng.randrange(0, 4), **{v: 2})))
        else:
            v = rng.choice(vars_)
            toks.append(('10', L(rng.randrange(0, 3), **{v: 1})))
    return toks


def tok_cells(toks, val):
    out = []
    for t in toks:
        if t[0] == '0':
            out.append(0)
        elif t[0] == '1':
            out += [1] * sx.ev(t[1], val)
        else:
            out += [1, 0] * sx.ev(t[1], val)
    return out


def rand_case(rng, vars_):
    """a random symbolic configuration; half of them are set up so that a shift rule fires at once"""
    left = rand_tokens(rng, vars_)
    right = rand_tokens(rng, vars_)
    st = rng.choice('ABCDEF')
    r = rng.random()
    v = rng.choice(vars_)
    X = L(rng.randrange(0, 2), **{v: 1})
    if r < 0.15:          # R1
        st = 'C'
        right = [('1', L(rng.randrange(0, 3), **{v: 2}))] + right
    elif r < 0.35:        # R2b, with a random cell (or block) left of the crossed copies
        st = 'B'
        left = left + [('10', X)]
        right = [('0',)] * 3 + right
    elif r < 0.5:         # R2a
        st = 'C'
        left = left + [('10', X), ('1', L(2))]
        right = [('0',)] + right
    return left, st, right


def eng_case(strict):
    vars_ = ['u', 'v', 'w']
    left, st, right = rand_case(RNG, vars_)
    ex = sx.Exec(left, st, right)
    ex.strict = strict
    start_left, start_right = list(left), list(right)
    sub = {}
    nst = RNG.randrange(1, 300)
    halted = False
    try:
        for _ in range(nst):
            while True:
                try:
                    ex.step()
                    break
                except sx.NeedSplit as ns:
                    s1 = {ns.var: L(0)} if RNG.random() < 0.4 else {ns.var: L(1, **{ns.var: 1})}
                    ex = ex.subst(s1)
                    sub = sx.compose(sub, s1)
    except sx.Halt:
        halted = True
    fired = sum(ex.uses.values())
    # instantiate: fresh values for the leaf variables
    fv = {v: RNG.randrange(0, 6) for v in vars_}
    orig = {v: sx.ev(sub[v], fv) if v in sub else fv[v] for v in vars_}
    lc, rc = tok_cells(start_left, orig), tok_cells(start_right, orig)
    tape = {i - len(lc): b for i, b in enumerate(lc + rc)}
    steps = sx.ev(ex.steps, fv)
    kind, ns_, tape, pos, st2 = run_dict(tape, 0, st, steps)
    el = tok_cells(ex.left, fv)
    er = tok_cells(list(reversed(ex.right)), fv)
    ok = (kind == 'OK') and st2 == ex.st
    if ok:
        # compare all cells relative to the head
        span = range(-len(el) - 5, len(er) + 5)
        for off in span:
            want = el[len(el) + off] if -len(el) <= off < 0 else (er[off] if 0 <= off < len(er) else 0)
            if tape.get(pos + off, 0) != want:
                ok = False
                break
        # and nothing nonzero outside the engine's words
        if ok:
            for p_, b in tape.items():
                if b and not (pos - len(el) <= p_ < pos + len(er)):
                    ok = False
                    break
    if halted:
        # the engine stopped before a '---'; the direct run must also stop there after `steps` steps
        ok = kind == 'OK' and TABLE[(st2, tape.get(pos, 0))] is None
    return ok, fired, (start_left, st, start_right, sub, fv)


def check_ENG():
    n_cases = scale(400, 2000, 10000)
    fired = 0
    for _ in range(n_cases):
        ok, f, info = eng_case(True)
        fired += f > 0
        rec('ENG.differential_random_symbolic_tapes', ok, info)
    rec('ENG.a_shift_rule_fired_in_>=30%_of_cases', fired >= 0.3 * n_cases, (fired, n_cases))
    # teeth: the same test catches an executor that drops R2a/R2b's 0-on-the-left condition
    bad = sum(not eng_case(False)[0] for _ in range(n_cases))
    rec('ENG.teeth_unsound_variant_is_caught', bad > 0, bad)

# proof.md Appendices A and B: one row per trip, 'trip  steps  word-after-G'; row 0 is the start word
APPENDIX = {'TE': 'Appendix A', 'TO': 'Appendix B'}
# the described structure (proof.md 3.4): rows 12j (j >= 1): first block n + base + 48(j-1), and the second
# variable block = shrink - 8(j-1); in TO the b and c blocks merge first at trip 10
S_PATTERN = {'TE': (52, 'b-2'), 'TO': (48, 'b+c-3')}


def appendix_rows(rule):
    t = open(os.path.join(HERE, 'proof.md')).read()
    i = t.index('\n## %s.' % APPENDIX[rule])
    blocks = t[i:].split('```')
    blk = blocks[3]                       # [1] is the rule statement, [3] the table
    lines = [ln for ln in blk.split('\n') if ln.strip()]
    assert lines[0].split()[0] == 'trip', lines[0]
    rows = []
    for ln in lines[1:]:
        m = re.fullmatch(r'\s*(\d+)\s+(\S+)\s+(\S.*\S)\s*', ln)
        assert m, ln
        rows.append((int(m.group(1)), m.group(2), m.group(3)))
    return rows


def var_tokens(toks):
    """the 1-blocks of a parsed word whose exponent has a variable, as (index, form)"""
    return [(i, e) for i, (sym, e) in enumerate(toks) if any(k != '' and c for k, c in e.items())]


def check_COL():

    for rule in ('TE', 'TO'):
        R = RULES[rule]
        dom = R['dom']
        forms = K_forms(dom)
        try:
            rows = appendix_rows(rule)
        except (ValueError, IndexError, AssertionError) as e:
            rec('COL.%s_table_parsed_from_proof_md' % rule, False, e)
            continue
        rec('COL.%s_table_parsed_from_proof_md' % rule, True)
        rec('COL.%s_rows_0_to_%d' % (rule, R['trips']), [r[0] for r in rows] == list(range(R['trips'] + 1)),
            [r[0] for r in rows][:5])
        words = [parse_word(GWORD_RUNS + ' ' + r[2]) for r in rows]
        words_all[rule] = (rows, words)
        rec('COL.%s_row0_is_K_start' % rule,
            sx.eqruns(sx.norm(word_tokens(words[0], forms)), sx.norm(sx.Ktoks(*[forms[v] for v in VARS]))))
        rec('COL.%s_last_row_is_K_end' % rule,
            sx.eqruns(sx.norm(word_tokens(words[-1], forms)), sx.norm(sx.Ktoks(*end_forms(rule, forms)))))
        tot = {'': 0}
        for j in range(1, len(rows)):
            claimed = parse_lin(rows[j][1])
            tot = add(tot, claimed)
            ex = sx.Exec([], 'F', word_tokens(words[j - 1], forms), rules=('R1', 'R2b'))
            lv = sx.explore(ex, 2, lambda sub, j=j: word_tokens(
                words[j], {v: subst_expr(forms[v], sub) for v in VARS}))
            rec('COL.%s_every_trip_by_executor_one_branch' % rule, len(lv) == 1 and lv[0].ok and not lv[0].sub,
                (j, [(l.sub, l.ok, fmt_got(l)) for l in lv]))
            for l in lv:
                o = leaf_orig(l.sub, forms)
                rec('COL.%s_every_trip_step_formula' % rule, sx.eqexpr(subst_expr(claimed, o), l.ex.steps),
                    (j, show(subst_expr(claimed, o)), show(l.ex.steps)))
                rec('COL.%s_no_D0' % rule, 'D0' not in l.ex.trans, j)
            # structure claims of proof.md 3.4
            rec('COL.%s_every_trip_has_7n' % rule, claimed.get('n') == 7, (j, rows[j][1]))
            rec('COL.%s_variable_coefficients_multiple_of_7' % rule,

        rec('COL.%s_step_column_sums_to_rule_formula' % rule, sx.eqexpr(tot, parse_lin(R['steps'])), show(tot))

            # right of the d block the word is constant: the last variable block is the d block, and d only

def direct_rule(rule, val):
    return ('F', 0, 2 * RULES[rule]['trips'], 10 ** 10, Kval(val))


def check_result(res, exp_word, exp_steps):
    k, s, q, h, w = res
    return k == 'REC' and q == 'F' and h == 0 and rstrip0(w) == rstrip0(exp_word) and s == exp_steps

def check_TEETH():
    # one value below each lower bound of S (same parity): the claimed result fails.  (The bounds of TE, TO, T6,
    # T8 are sufficient, not minimal: proof.md 3.4 claims nothing about them.)
    for rule in ('S',):
        R = RULES[rule]
        for v in VARS:
            x = R['dom'][v]
            if isinstance(x, int) or v == 'n':
                continue
            fails = 0
            vals = []
            for _ in range(scale(3, 6, 12)):
                val = dom_sample_rand(R['dom'], 60, RNG)
                val[v] = x[1] - 2
                vals.append(val)
            res = krun([direct_rule(rule, val) for val in vals])
            for val, r in zip(vals, res):
                fails += not check_result(r, Kval(apply_rule(rule, val)), rule_steps(rule, val))
            rec('TEETH.%s_%s>=%d_sharp' % (rule, v, x[1]), fails == len(vals), (fails, len(vals)))

def even(n, x):
    return {'n': n, 'a': x, 'b': (x + 76) // 2, 'c': 35, 'd': 18}


def double_cycle(n, x, steps_out=None):
    """apply the rules S^((x-4)/8), TE, S^((x-52)/8), TO with domain checks; return (Even params, steps, ok)"""
    val = even(n, x)
    total = 0
    ok = True
    seq = ['S'] * ((x - 4) // 8) + ['TE'] + ['S'] * ((x - 52) // 8) + ['TO']
    for r in seq:
        ok &= in_dom(RULES[r]['dom'], val)
        total += rule_steps(r, val)
        val = apply_rule(r, val)
    return val, total, ok, len(seq)

        rec('DC.progress', tot >= 1)

def Kr(n, a, b, c, d):
    """proof.md 8: Kr n a b c d = K(2n+1, 2a, 2b, 2c+1, 2d), as a parameter dict"""
    return {'n': 2 * n + 1, 'a': 2 * a, 'b': 2 * b, 'c': 2 * c + 1, 'd': 2 * d}


def Ev(N, X):
    return Kr(N, 4 * X + 26, 2 * X + 32, 17, 9)


PLAN_LEMMAS = {
    # name: (rule, start args, end args) as functions of the lemma's arguments
    'S_step': ('S', lambda n, a, b, c, d: Kr(n, 5 + a, 3 + b, 3 + c, 1 + d),
               lambda n, a, b, c, d: Kr(24 + n, 1 + a, 5 + b, 3 + c, 3 + d), 5),
    'TE': ('TE', lambda n, b, d: Kr(n, 2, 44 + b, 17, 10 + d), lambda n, b, d: Kr(278 + n, 1 + b, 41, 35 + d, 9), 3),
    'TO': ('TO', lambda n, b, c, d: Kr(n, 1, 41 + b, 18 + c, 9 + d),
           lambda n, b, c, d: Kr(294 + n, 14 + b + c, 40 + d, 17, 9), 4),
    'T6': ('T6', lambda n, b, c, d: Kr(n, 3, 41 + b, 17 + c, 9 + d),
           lambda n, b, c, d: Kr(296 + n, 13 + b + c, 40 + d, 17, 9), 4),
    'T8': ('T8', lambda n, b, d: Kr(n, 4, 44 + b, 17, 10 + d), lambda n, b, d: Kr(270 + n, 7 + b, 43, 33 + d, 9), 3),
}


def S_iter_start(k, n, a, b, c, d):
    return Kr(n, 4 * k + 1 + a, 3 + b, 3 + c, 1 + d)


def S_iter_end(k, n, a, b, c, d):
    return Kr(24 * k + n, 1 + a, 2 * k + 3 + b, 3 + c, 2 * k + 1 + d)
