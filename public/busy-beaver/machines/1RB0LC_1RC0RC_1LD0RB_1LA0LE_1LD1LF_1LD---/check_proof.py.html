# A configuration is (cells: dict pos->bit, head pos, state).  For comparison we render it as

def render(t, h, s):
    ones = [i for i, x in enumerate(t) if x]
    lo = min(ones + [h]); hi = max(ones + [h])
    return ''.join(str(t[i]) for i in range(lo, hi + 1)), h - lo, s

def parse_conf(w):
    """'X <E Y' (head on the last cell of X) or 'X E> Y' (head on the first cell of Y); spaces ignored;
    '0^inf' allowed at either end (ignored).  Returns (cells, head offset, state), blank-padded by 4."""
    w = w.replace('0^inf', '').replace(' ', '')
    m = re.search(r'<([A-F])', w)
    if m:
        L, R = w[:m.start()], w[m.end():]; h = len(L) - 1
    else:

        L, R = w[:m.start()], w[m.end():]; h = len(L)
    return '0000' + L + R + '0000', h + 4, ST.index(m.group(1))

def rnd(r, lo, hi):

        okM += same(run_from(Kword(L, D, a, b), n), parse_conf(want))
    for _ in range(N):
        D = [(rnd(r, 2, 20), 1, rnd(r, 1, 20)) for _ in range(r.randint(1, 6))]
        a, b = rnd(r, 1, 40), rnd(r, 0, 40)

    for _ in range(N):

                                   [(rnd(r, 1, 9), r.randint(0, 1), rnd(r, 1, 9)) for _ in range(r.randint(0, 2))])

    report('5', okM == okO == okP == N, 'M %d/%d, O %d/%d, P %d/%d on random K-configs (exact tape, head, state '

# ---------------------------------------------------------------- 6. P chain
def P_chain(L, S, a, b):
    """the intermediate configurations of proof.md 3.3, with cumulative step counts"""
    u = [d[0] for d in S]; v = [d[2] for d in S]; k = len(S)
    Yp = ''.join('01' * u[i] + '10' * (v[i] - 1) + '11' for i in range(1, k))   # swept d_{p+1..m}
    Y3 = ''.join('10' * (u[i] - 1) + '1' + '10' * (v[i] - 1) + '000' for i in range(1, k))
    tail = '01' * a + '011' + '01' * (b + 2)
    links = []
    n = 4 * b + 10
    links.append(('phase 1', L + dig(S) + '10' * (a + 1) + '0' + '10' * (b + 1) + ' B> 0^inf', n))
    n += 2 * a + 2 * b + 8 + sum(2 * u[i] + 2 * v[i] for i in range(1, k))
    links.append(('phase 2 (Lb on d_{p+1..m})', L + '10' + '01' * v[0] + ' <E ' + Yp + tail, n))
    n += 8 * v[0] + 1
    links.append(('Lt on d_p', L[:-1] + '01' + '10' * (v[0] - 1) + '000 B> ' + Yp + tail, n))
    n += sum(2 * u[i] + 6 * v[i] - 4 for i in range(1, k))
    links.append(('Ro on d_{p+1..m}', L[:-1] + '01' + '10' * (v[0] - 1) + '000' + Y3 + ' B> ' + tail, n))
    want, steps = P_rule(L, S, a, b)
    n += 2 * a + 8 * b + 30 + 2 * a + 1 + 2 * a + 6 * b + 27
    links.append(('phases 5-7 (as in O)', want, n))
    return links, steps

def check_P_chain(r, nrand):

    for Lc in ['1', '1001', '100101', '10011001', '1001' + dig([(2, 0, 1)]) + '01']:
        for v0 in (1, 2, 3):

# ---------------------------------------------------------------- 11/12. the general rule G (proof.md 3.5, 3.6)
# K(L; a, b): phase 1 and phase 2 as in M; the sweep then crosses the maximal swept suffix of L (Lb on bit-1
# digits with u >= 2, Lb1 on a u = 1 digit whose left cell is 0), and turns on the text X left of it.  C-exit
# turns are followed by Rb on the swept digits and M's right end; B-exit turns by Ro and O's phases 5-7.
def _seg(S, i, j):
    """S[i:j]; cells left of S are blank"""
    if i >= 0:
        return S[i:j]
    return '0' * (min(j, 0) - i) + (S[:j] if j > 0 else '')

def _runback(S, j, w):
    k = 0
    while _seg(S, j - 2, j) == w:
        k += 1; j -= 2
    return k, j

def g_turn(S, j):
    """turn on X = S[:j].  (name, keep, W, cost, exit): X is replaced by S[:keep] + W"""
    v, i = _runback(S, j, '01')
    if v >= 1 and _seg(S, i - 2, i) == '00':
        return 'Lf', i - 2, '10' + '01' * (v - 1), 4 * v + 1, 'C'      # 00 (01)^v      -> 10 (01)^(v-1) C> 11
    if _seg(S, j - 4, j) == '1100':
        return 'T3S2', j - 4, '01', 5, 'C'                             # 1100           -> 01 C> 11
    if _seg(S, j - 2, j) == '00':
        v, i = _runback(S, j - 2, '01')
        if v >= 1 and _seg(S, i - 2, i) == '00':
            return 'Jc', i - 2, '10' + '01' * v, 4 * v + 5, 'C'        # 00 (01)^v 00   -> 10 (01)^v C> 11
    if _seg(S, j - 3, j) == '000':
        return 'T2', j - 3, '100', 5, 'B'                              # 000            -> 100 B>
    return None, xclass(S[:j]), None, None, None

def xclass(X):
    """proof.md 3.6: the class of a left text X (blank-padded) that does not end in 10"""
    X = '0' * 6 + X
    v, i = _runback(X, len(X), '01')
    if v == 0 and X.endswith('11'):
        return 'H1'
    if v >= 1:
        e = X[:i]
        return 'Lf' if e.endswith('00') else 'H2' if e.endswith('111') else 'U1' if e.endswith('011') else '?'
    if X.endswith('000'):
        return 'T2'
    if X.endswith('1100'):
        return 'T3S2'
    if X.endswith('0100'):
        w, i = _runback(X, len(X) - 4, '01')
        e = X[:i]
        return 'Jc' if e.endswith('00') else 'U2' if e.endswith('10') else 'U3' if e.endswith('11') else '?'
    return '?'

def g_split(L):
    """maximal swept suffix of L: (j, swept digits left to right, Lt-turn or None)"""
    j = len(L); swept = []; t = None
    while True:
        v, i = _runback(L, j, '01')
        u, i2 = _runback(L, i, '10')
        if u == 0 or v == 0:
            break
        if u >= 2 or _seg(L, i2 - 1, i2) == '0':
            swept.append((u, v)); j = i2
            if u == 1:
                break                         # Lb1: exits C before a 0, the next cell is not a digit end
            continue
        t = ('Lt', i2 - 1, '01' + '10' * (v - 1) + '000', 8 * v + 1, 'B')  # 1 10 (01)^v -> 01 (10)^(v-1) 000 B>
        break
    return j, swept[::-1], t

def g_predict(L, a, b):
    """the general rule G.  returns ((type, turn, k), predicted K-word, steps) or (None, class, None)"""
    if a < 1:
        return None, 'a = 0', None
    j, swept, t = g_split(L)
    if t is None:
        t = g_turn(L, j)
        if t[0] is None:
            return None, t[1], None
    name, keep, W, cost, ex = t
    head = L[:keep] if keep > 0 else ''
    n = (4 * b + 10) + (2 * a + 2 * b + 8) + sum(2 * u + 2 * v for u, v in swept) + cost
    if ex == 'C':
        n += sum(6 * u + 2 * v for u, v in swept) + 6 * a + 2 * b + 13
        out = head + W + ''.join('10' * u + '00' + '01' * (v - 1) for u, v in swept) + '10' * (a + 1) + '000' + '10' * (b + 2)
        typ = 'M*'
    else:
        Z = head + W + ''.join('10' * (u - 1) + '1' + '10' * (v - 1) + '000' for u, v in swept)
        if not Z.endswith('000'):
            return None, 'O*-turn with no swept digit', None
        n += sum(2 * u + 6 * v - 4 for u, v in swept) + 6 * a + 14 * b + 58
        out = Z[:-3] + '100' + '10' * (a - 1) + '1' + '10' * (b + 3) + '000' + '10'
        typ = 'O*'
    return (typ, name, len(swept)), out.lstrip('0') + ' B> 0^inf', n

def check_general(r, N):

def check_sweep_classes(r):

# ---------------------------------------------------------------- K-config parsing (kdump output)
TOK = {'a': '10', 'z': '00', 'o': '01', 'X': '11', '1': '1'}
def load_k(lines):
    out = []
    for line in lines:
        p = line.split()
        if not p or not p[0].isdigit():
            continue
        out.append((int(p[0]), int(p[1]), int(p[2]), [(x[0], int(x[1:])) for x in p[3:]]))
    return out

def text_of(runs):
    return ''.join(TOK[c] * n for (c, n) in reversed(runs))

def digits_of(runs):
    toks = [[c, n] for c, n in runs]
    D = []; i = 0
    while i < len(toks):
        v = 0; beta = 1; u = 0; j = i
        if toks[j][0] == 'o':
            v = toks[j][1]; j += 1
        if j < len(toks) and toks[j][0] == 'z' and toks[j][1] == 1:
            beta = 0; j += 1
        if j < len(toks) and toks[j][0] == 'a':
            u = toks[j][1]; j += 1
        if j == i or (v == 0 and beta == 1):
            break
        D.append((u, beta, v)); i = j
    return toks[i:], D[::-1]

def predict(cfg):
    """which of M / O / P applies (proof.md 3), and its predicted K-word and step count"""
    st, a, b, runs = cfg
    pre, D = digits_of(runs)
    pre_txt = ''.join(TOK[c] * n for (c, n) in reversed(pre))
    if not D:
        return None, 'no digits'
    m = len(D); k = 0
    while k < m and D[m - 1 - k][1] == 1:
        k += 1
    for i in range(m - 1, m - 1 - k, -1):
        u, _, v = D[i]
        if v < 1:
            return None, 'swept digit with v = 0'
        if u == 1:
            L = pre_txt + dig(D[:i])
            if i == 0 and pre_txt == '':
                if v == 1:
                    break
                return None, 'far-left wall 10 (01)^v, v >= 2'
            if not L.endswith('1'):
                return None, 'u = 1 digit with left neighbour 0'
            return P_rule(L, D[i:], a, b), 'P'
        if u == 0:
            return None, 'swept bit-1 digit with u = 0'
    if k < m:
        js = m - 1 - k
        if D[js][2] < 1:
            return None, 'carry reaches bit-0 digit with v = 0'
        if a < 1:
            return None, 'a = 0'
        return M_rule(pre_txt + dig(D[:js]), D[js:], a, b), 'M'
    if pre_txt == '' and D[0] == (1, 1, 1):
        if m == 1:
            return None, 'O with no digits'
        return O_rule(D[1:], a, b), 'O'
    return None, 'sweep enters the prefix'

def kword_of(cfg):
    st, a, b, runs = cfg
    return text_of(runs) + '10' * a + '000' + '10' * b + ' B> 0^inf'

def fast_run(text_with_head, n):
    """run n steps from 'text B> 0^inf' on a bytearray; return render() triple"""
    txt = text_with_head.replace(' B> 0^inf', '').lstrip('0')
    pad = len(txt) + 64 + n // 4
    t = bytearray(pad) + bytearray(int(c) for c in txt) + bytearray(pad)
    h = pad + len(txt); s = 1
    for _ in range(n):
        tr = TAB[s][t[h]]
        if tr is None:
            return None
        t[h] = tr[0]; h += tr[1]; s = tr[2]
    lo = t.find(1); hi = t.rfind(1)
    lo = min(lo, h); hi = max(hi, h)
    return ''.join('01'[x] for x in t[lo:hi + 1]), h - lo, s

# ---------------------------------------------------------------- 8/9. census and coverage
# the greedy cover uses the 23 windows of the earlier stage: T3j (= T3 then one C0 step) is left out, because
# longest-first greedy would also take it where T3 is followed by S4c, and no window starts with the D1 step
# that then follows
ATOM = sorted(((len(parse_window(l)[0]), n_, parse_window(l), s_) for n_, (l, _, s_) in LEM.items() if n_ != 'T3j'),
              key=lambda x: -x[0])
def cover(text_with_head, nsteps):
    """greedy decomposition into atomic windows; returns (#lemma applications, #uncovered steps, end render)"""
    txt = text_with_head.replace(' B> 0^inf', '').lstrip('0')
    pad = len(txt) + 4096
    t = bytearray(pad) + bytearray(int(c) for c in txt) + bytearray(pad)
    h = pad + len(txt); s = 1; done = 0; apps = 0; unk = 0
    pats = [(L, name, [int(x) for x in c], off, ST.index(st), n) for (L, name, (c, off, st), n) in ATOM]
    while done < nsteps:
        hit = None
        for (L, name, cells, off, st, n) in pats:
            if st != s or done + n > nsteps:
                continue
            b0 = h - off
            if b0 < 0 or b0 + L > len(t):
                continue
            okp = True
            for i in range(L):
                if t[b0 + i] != cells[i]:
                    okp = False; break
            if okp:
                hit = n; break
        n = hit if hit else 1
        if not hit:
            unk += 1
        for _ in range(n):
            tr = TAB[s][t[h]]
            if tr is None:
                return apps, unk + 1, None
            t[h] = tr[0]; h += tr[1]; s = tr[2]
        done += n; apps += 1
    lo = min(t.find(1), h); hi = max(t.rfind(1), h)
    return apps, unk, (''.join('01'[x] for x in t[lo:hi + 1]), h - lo, s)

def check_census_cover(kd, T1, T2, expect):
    cnt = collections.Counter(); bad = 0
    for c1, c2 in zip(kd, kd[1:]):
        pr, why = predict(c1)
        if pr is None:
            cnt['other: ' + why] += 1; continue
        if pr[1] == c2[0] - c1[0] and same(parse_conf(pr[0]), parse_conf(kword_of(c2))):
            cnt[why] += 1
        else:
            bad += 1
    other = sum(v for k, v in cnt.items() if k.startswith('other'))
    got = (cnt['M'], cnt['P'], cnt['O'], other, bad)
    detail = ', '.join('%s %d' % (k, v) for k, v in sorted(cnt.items(), key=lambda x: -x[1]))
    # the general rule G: every transition, by turn
    gcnt = collections.Counter(); gbad = 0; gnone = collections.Counter()
    for c1, c2 in zip(kd, kd[1:]):
        L = text_of(c1[3]).lstrip('0')
        k, want, n = g_predict(L, c1[1], c1[2])
        if k is None:
            gnone[want] += 1; continue
        if n == c2[0] - c1[0] and want.replace(' B> 0^inf', '') == text_of(c2[3]).lstrip('0') + '10' * c2[1] + '000' + '10' * c2[2]:
            gcnt[k[1]] += 1
        else:
            gbad += 1
    gdet = ', '.join('%s %d' % (t, gcnt[t]) for t in ('Lf', 'T3S2', 'Jc', 'T2', 'Lt'))
    gok = gbad == 0 and not gnone and sum(gcnt.values()) == len(kd) - 1
    if expect:
        gok = gok and tuple(gcnt[t] for t in ('Lf', 'T3S2', 'Jc', 'T2', 'Lt')) == (22353, 8, 2, 37, 35)
    gtxt = ('; general rule G: %d of %d transitions predicted exactly (%s), unpredicted %d, mismatches %d'
            % (sum(gcnt.values()), len(kd) - 1, gdet, sum(gnone.values()), gbad))
    if expect:
        report('8', got == expect and gok, 'census of K->K transitions to %.0e: %s; mismatches %d (expected M/P/O/other '
               '= %s)%s' % (T2, detail, bad, '/'.join(map(str, expect[:4])), gtxt))
    else:
        report('8', bad == 0 and gok, 'census to %.0e: %s; mismatches %d%s' % (T2, detail, bad, gtxt))
    # coverage
    nall = nother = unk_tot = endbad = 0
    for c1, c2 in zip(kd, kd[1:]):
        pr, why = predict(c1)
        if c1[0] < T1 or why != 'M':
            apps, unk, end = cover(kword_of(c1), c2[0] - c1[0])
            unk_tot += unk
            endbad += not same(end, parse_conf(kword_of(c2)))
            if c1[0] < T1:
                nall += 1
            else:
                nother += 1
    report('9', unk_tot == 0 and endbad == 0,
           'coverage by 23 of the atomic windows (T3j left out): all %d transitions before step %.0e and the %d '
           'non-M transitions after it (to %.0e); uncovered steps %d; end-config mismatches %d.  This shows only '
           'that the run uses no other local patterns; it is not evidence for non-halting' % (nall, T1, nother, T2, unk_tot, endbad))
