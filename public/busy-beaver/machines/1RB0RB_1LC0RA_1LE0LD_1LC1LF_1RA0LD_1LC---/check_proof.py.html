def relabel(tab, perm, mirror):
    """machine string of table tab with parent state s renamed perm[s] (and L/R swapped if mirror)."""
    rows = [None] * 6
    for s in range(6):
        cells = []
        for sym in (0, 1):
            tr = tab[s][sym]
            if tr is None:
                cells.append('---')
            else:
                w, d, t = tr
                cells.append('%d%s%s' % (w, 'R' if (d == 1) != mirror else 'L', chr(65 + perm[t])))
        rows[perm[s]] = ''.join(cells)
    return '_'.join(rows)

def maps_to(tab, target):
    return [(''.join(chr(65 + p) for p in perm), mir, chr(65 + perm.index(0)))
            for perm in itertools.permutations(range(6)) for mir in (False, True)
            if relabel(tab, list(perm), mir) == target]

def run(cells, head, state, n, tab=None):
    """n steps on THIS table (or tab). Returns (tape_str, head_idx, state, min_head, max_head), the last two
    relative to cells[0]."""
    tab = TT if tab is None else tab
    pad = n + 8
    t = bytearray(pad) + bytearray(int(c) for c in cells) + bytearray(pad)
    h = head + pad; st = ord(state) - 65; mn = mx = h
    for _ in range(n):
        tr = tab[st][t[h]]
        if tr is None:
            raise P.Halt()
        t[h] = tr[0]; h += tr[1]; st = tr[2]
        if h < mn: mn = h
        if h > mx: mx = h
    return ''.join('01'[x] for x in t), h, chr(st + 65), mn - pad, mx - pad

def render(D, a, b):

def parse_K(txt):
    """txt = the cells from the leading 1 up to the cell left of the head. Returns (D, a, b) if
    txt = 1001 d_1..d_m (10)^a 000 (10)^b with canonical-shape digits (v >= 1), else None. The parse is unique:
    b and a are forced by stripping 10s from the right (a digit ends in 01), the digits by greedy reading."""
    if not txt.startswith('1001'):
        return None
    s = txt; b = 0
    while s.endswith('10'):
        s = s[:-2]; b += 1
    if not s.endswith('000'):
        return None
    s = s[:-3]; a = 0
    while s.endswith('10') and len(s) > 4:
        s = s[:-2]; a += 1
    mid = s[4:]; D = []; i = 0
    while i < len(mid):
        u = 0
        while mid.startswith('10', i): u += 1; i += 2
        beta = 1
        if mid.startswith('00', i): beta = 0; i += 2
        v = 0
        while mid.startswith('01', i): v += 1; i += 2
        if v == 0:
            return None
        D.append((u, beta, v))
    if P.render(D, a, b)[0] != txt:
        return None
    return D, a, b

def scan_K(tab, start, kq, T):
    """every K-configuration in steps 0..T from the blank tape: head state kq on a cell with only blanks
    from there to the right, only blanks left of the leading 1, the text in between parsing as a K."""
    n = T + 16
    tape = bytearray(2 * n); h = n; s = ord(start) - 65; q = ord(kq) - 65; out = []
    for t in range(T + 1):
        if t > 0:
            tr = tab[s][tape[h]]
            if tr is None:
                return out, t
            tape[h] = tr[0]; h += tr[1]; s = tr[2]
        if s == q and not any(tape[h:]):
            f = tape.find(1)
            if 0 <= f < h:
                k = parse_K(''.join('01'[x] for x in tape[f:h]))
                if k:
                    out.append((t, k))
    return out, None

def derived(D, a, b):

# ---------------------------------------------------------------- R lockstep (Lemma R on the real machines)
def check_lockstep(N):
    n = N + 8
    t1 = bytearray(2 * n); t2 = bytearray(2 * n); h1 = h2 = n
    s1 = 0; s2 = 1                                         # this machine from A; #11 from B
    tau = [ord(TAU[chr(65 + i)]) - 65 for i in range(6)]
    tw, pa = TT, P.TT
    ok = 0
    for _ in range(N):
        if s1 != tau[s2]:
            break
        a = tw[s1][t1[h1]]; b = pa[s2][t2[h2]]
        if a is None or b is None or a[0] != b[0] or a[1] != b[1]:
            break
        t1[h1] = a[0]; h1 += a[1]; s1 = a[2]
        t2[h2] = b[0]; h2 += b[1]; s2 = b[2]
        ok += 1
    same = ok == N and t1 == t2 and h1 == h2 and s1 == tau[s2]
    return report('R lockstep', ok + same, N + 1, '(this machine from blank = tau(#11 from 0^inf B> 0^inf) at every '
                  'step to %d: state, symbol, move; final tapes identical)' % N)


# ---------------------------------------------------------------- L lemmas
def ren_window(w):
    i = w.find('>')
    if i > 0 and w[i - 1] in 'ABCDEF':
        return w[:i - 1] + TAU[w[i - 1]] + w[i:]
    i = w.find('<')
    return w[:i + 1] + TAU[w[i + 1]] + w[i + 2:]

def lemma_run(lhs, rhs, n, tab):
    c, h, st = P.parse_window(lhs); c2, h2, st2 = P.parse_window(rhs)
    t = [int(x) for x in c]; s = ord(st) - 65; trs = []; fr = []
    for _ in range(n):
        if not (0 <= h < len(t)) or tab[s][t[h]] is None:
            return False, trs, fr
        trs.append(chr(s + 65) + str(t[h]))
        if s == 5: fr.append(t[h])
        w, d, s = tab[s][t[h]]; t[h] = w; h += d
    good = len(c) == len(c2) and ''.join(map(str, t)) == c2 and h == h2 and s == ord(st2) - 65
    return good, trs, fr

def check_lemmas(verbose=True):
    ok = 0; fs = []; lines = []
    for name, (lhs, rhs, n) in P.LEM.items():
        L2, R2 = ren_window(lhs), ren_window(rhs)

        pg, ptrs, _ = lemma_run(lhs, rhs, n, P.TT)
        good = good and pg and [TAU[x[0]] + x[1] for x in ptrs] == trs
        ok += good
        fs += [(name, x) for x in fr]

def run_chain(c0, ln, pad=64):

    l, mo, st, r = c0

    t = bytearray(pad) + bytearray(int(x) for x in l + r) + bytearray(pad + n)
    h = pad + (len(l) if mo == '>' else len(l) - 1); s = ord(TAU[st]) - 65; mn = h; tot = 0
    for (nm, k, (l2, mo2, st2, r2)) in ln:
        for _ in range(P.LEM[nm][2] * k):

            if tr is None:
                return False, tot, None, None
            t[h] = tr[0]; h += tr[1]; s = tr[2]; mn = min(mn, h); tot += 1

def check_closure(Tmax):

    tape = bytearray(Lpad + Rpad); tab = TT

    tr01 = bytes.maketrans(b'01', b'\x00\x01')
    def go(n):
        t_ = tape; hh = state['h']; ss = state['s']
        for _ in range(n):
            tr = tab[ss][t_[hh]]
            if tr is None:
                raise P.Halt()
            t_[hh] = tr[0]; hh += tr[1]; ss = tr[2]
        state['h'] = hh; state['s'] = ss
    def matches(D, a, b):
        s_, _, _ = render(D, a, b)
        if state['s'] != ord(KQ) - 65:
            return False
        seg = s_.encode().translate(tr01); hh = state['h']
        return tape[hh - len(seg):hh] == seg and tape.count(1) == s_.count('1')
    go(T0)
    (D, a, b), t = BASE, T0
    n = ok = int(matches(D, a, b)); names = Counter(); cls = Counter(); drops = Counter()
    while True:
        c = P.inv(D, a, b); cls[c] += 1
        if c is None:
            print('  closure: INV fails at', t); ok -= 1; break
        name, D2, a2, b2, st, info = P.mstep(D, a, b)
        if t + st > Tmax:

        go(st)
        D, a, b, t = D2, a2, b2, t + st
        g = matches(D, a, b)
        n += 1; ok += g
        names[name if name == 'O' or int(name[1:]) < 6 else 'M6+'] += 1
        if info.get('drop'): drops[info['drop']] += 1
        if not g:
            print('  closure FAIL at step', t, name); break
    return report('D closure', ok, n, '(K at 732 + every macro step of the real run to step %d: %s; drops %s; '
                  'INV classes %s; last K: m=%d, a=%d, b=%d)' % (t, dict(names), dict(drops),

# ---------------------------------------------------------------- X coupling (explanation, not a premise)
def _rel(x, y):
    (D1, a1, b1), (D2, a2, b2) = x, y
    if a1 != a2 or b1 != b2 or len(D1) != len(D2):
        return None
    out = []
    for j, (d1, d2) in enumerate(zip(D1, D2)):
        if d1[1] != d2[1]: return None
        if d1[0] != d2[0]: out.append((j + 1, 'u', d1[0] - d2[0]))
        if d1[2] != d2[2]: out.append((j + 1, 'v', d1[2] - d2[2]))
    return tuple(out)

def coupling(x, tx, y, ty, N, delta):
    """from the first macro step k at which the states differ in exactly one top-digit exponent by delta,
    check that this persists and that the same rule fires, to N macro steps."""
    first = None; seen = Counter(); ok = tot = 0
    for k in range(N):
        r_ = _rel(x, y)
        good = r_ is not None and len(r_) == 1 and r_[0][2] == delta and r_[0][0] <= 2
        if good and first is None:
            first = (k, tx, ty)
        n1, D, a, b, s1, _ = P.mstep(*x); n2, D2, a2, b2, s2, _ = P.mstep(*y)
        if first is not None:
            tot += 1; good = good and n1 == n2; ok += good
            seen[(r_[0][0], r_[0][1], P.inv(*x)) if r_ else None] += 1
            if not good:
                print('  coupling FAIL at macro step', k, x, y, r_); break
        x, y, tx, ty = (D, a, b), (D2, a2, b2), tx + s1, ty + s2
    return first, ok, tot, seen, tx - ty, tx

def _nocomment(s):
    out = []; depth = 0; i = 0
    while i < len(s):
        if s.startswith('(*', i): depth += 1; i += 2; continue
        if depth and s.startswith('*)', i): depth -= 1; i += 2; continue
        if not depth: out.append(s[i])
        i += 1
    return '\n'.join(x.rstrip() for x in ''.join(out).split('\n') if x.strip())

    tail = one(tail, 'start: 0^inf A> 0^inf  -->*  K((1,1,2); 8, 5)                       (step 483)',
               'start: 0^inf A> 0^inf  -->*  K((1,1,10); 8, 1)                      (step 732)')
    return HEADER + body[:j] + tail

def check_rocq_delta():
    ok = tot = 0
    def chk(c, msg):
        nonlocal ok, tot
        tot += 1; ok += bool(c)
        if not c: print('  rocq delta FAIL:', msg)
    src = open(os.path.join(PDIR, PARENT + '.v')).read()
    chk(hashlib.sha256(src.encode()).hexdigest() == PARENT_V_SHA, 'parent .v sha256 pinned')
    dv = os.path.join(HERE, 'rocq', 'delta_check.v')
    if not os.path.exists(dv):
        chk(False, 'rocq/delta_check.v missing')
        return report('V rocq delta', ok, tot)
    d = open(dv).read()
    chk(delta_text(src) == d, 'delta_check.v == delta(parent .v), byte for byte')
    a = _nocomment(src); b = _nocomment(d)
    a2 = a.replace(PARENT, TW).replace('[d1 1 2] 8 5', '[d1 1 10] 8 1').replace('([d1 1 2], 8, 5)', '([d1 1 10], 8, 1)')
    a2 = re.sub(r'\{\{([A-F])\}\}', lambda m_: '{{%s}}' % TAU[m_.group(1)], a2)
    chk(a2 == b, 'code (comments stripped) = parent code with the delta substitutions only')
    nl = sum(1 for x, y in zip(a.split('\n'), b.split('\n')) if x != y)
    chk(len(re.findall(r'TM_from_str', b)) == 1 and 'Lemma nonhalt: ~halts tm c0.' in b, 'one TM_from_str, nonhalt')
    bad = re.findall(r'\b(Admitted|admit|Axiom|Parameter|Hypothesis|Conjecture|Abort|Variable|Context)\b', b)
    chk(not bad, 'no assumptions: %s' % bad)
    lg = os.path.join(HERE, 'rocq', 'delta_check.out')
    log = open(lg).read() if os.path.exists(lg) else ''
    sha = hashlib.sha256(d.encode()).hexdigest()
    chk(log.startswith(sha) and 'exit 0' in log and 'Closed under the global context' in log
        and re.search(r'nonhalt\s*:\s*~\s*halts tm c0', log), 'rocq/delta_check.out = clean compile of this exact file')
    return report('V rocq delta', ok, tot, '(delta_check.v = #11\'s .v + proof.md 5 delta, byte for byte; code lines '
                  'changed: %d; compile record: sha256 match, exit 0, closed; NOT a check_kill run)' % nl)
